\section{Proof Architecture}

Throughout, \(T\) is a complete first-order theory in a countable
language \(L\), and \(T\) has infinite models unless explicitly stated
otherwise.

The proof is organized around three milestones.  First, uncountable
categoricity implies \(\omega\)-stability.  Second, uncountable
categoricity rules out Vaughtian pairs.  Third, \(\omega\)-stability
together with the absence of Vaughtian pairs yields uncountable
categoricity through a strongly minimal set and its dimension theory.
The Baldwin--Lachlan characterization packages these three steps, and
Morley's theorem follows immediately.

The blueprint distinguishes reusable model-theoretic infrastructure from
results specialized to one of these milestones.  In particular, the
definably-full construction belongs to the Vaughtian-pair argument,
Skolemization and indiscernibles belong to the \(\omega\)-stability
argument, and strong-minimal pregeometry belongs to the reverse
implication.

\section{Core Model-Theoretic Infrastructure}

\begin{definition}[Categoricity in a cardinal]
\label{def:categoricity}
\lean{Cardinal.Categorical}
\mathlibok
The theory \(T\) is \(\kappa\)-categorical if any two models of \(T\)
of cardinality \(\kappa\) are isomorphic.
\end{definition}

\begin{definition}[Embeddings, isomorphisms, and partial embeddings]
\label{def:embeddings}
\lean{FirstOrder.Language.Embedding, FirstOrder.Language.Equiv, FirstOrder.Language.PartialEmbedding}
\leanok
An \(L\)-embedding is an injective map preserving the interpretations of
all function, relation, and constant symbols of \(L\).  An isomorphism
is a bijective \(L\)-embedding.  If \(A\subseteq M\) and \(B\subseteq N\),
a map \(f:A\to B\) is a partial embedding when it preserves all
quantifier-free \(L\)-formulas with parameters from its domain.
\emph{Formalization note.}  The subset version is
\texttt{FirstOrder.Language.PartialEmbedding}, with scoped notation
\(A\hookrightarrow_{\!p}[L] B\).
\end{definition}

\begin{definition}[Partial elementary embeddings]
\label{def:partial-elementary-embedding}
\lean{FirstOrder.Language.PartialElementaryEmbedding}
\leanok
\uses{def:embeddings}
A partial embedding \(f:A\to B\) is partial elementary if every
first-order formula with parameters from its domain is preserved in
both directions.  We write the corresponding bundled map as
\(A\hookrightarrow_{\!pe}[L] B\).
\end{definition}

\begin{definition}[Elementary embeddings and substructures]
\label{def:elementary-maps}
\lean{FirstOrder.Language.ElementaryEmbedding, FirstOrder.Language.Substructure.IsElementary, FirstOrder.Language.ElementarySubstructure}
\mathlibok
\uses{def:embeddings}
An embedding \(j:M\to N\) is elementary if it preserves every
first-order formula.  If the inclusion \(M\hookrightarrow N\) is
elementary, we write \(M\preceq N\).
\end{definition}

\begin{theorem}[Tarski--Vaught test]
\label{thm:tarski-vaught-test}
\lean{FirstOrder.Language.Substructure.isElementary_of_exists, FirstOrder.Language.Substructure.toElementarySubstructure}
\mathlibok
\uses{def:elementary-maps}
Let \(M\) be a substructure of \(N\).  Then \(M\preceq N\) if and only
if every existential formula over parameters from \(M\) that has a
witness in \(N\) already has a witness in \(M\).
\end{theorem}

\begin{definition}[Counting formulas for definable sets]
\label{def:counting-formulas}
\lean{FirstOrder.Language.Formula.iExsAtLeast, FirstOrder.Language.Formula.iExsAtMost, FirstOrder.Language.Formula.iExsExactly, FirstOrder.Language.Formula.existsExactlyLeft, FirstOrder.Language.Formula.realize_iExsAtLeast, FirstOrder.Language.Formula.realize_iExsAtMost, FirstOrder.Language.Formula.realize_iExsExactly, FirstOrder.Language.Formula.realize_existsExactlyLeft}
\leanok
For a formula \(\varphi(\bar x,\bar y)\) and \(n\in\mathbb N\), there
are formulas
\[
  (\exists^{\geq n}\bar x)\,\varphi,\qquad
  (\exists^{\leq n}\bar x)\,\varphi,\qquad
  (\exists^{=n}\bar x)\,\varphi
\]
asserting that the corresponding definable set has at least, at most, or
exactly \(n\) elements.
\end{definition}

\begin{lemma}[Definable-set cardinalities under elementary embeddings]
\label{lem:elementary-embedding-definable-set-cardinality}
\lean{FirstOrder.Language.ElementaryEmbedding.realizations_embedding, FirstOrder.Language.ElementaryEmbedding.mk_realizations_le, FirstOrder.Language.ElementaryEmbedding.encard_realizations_eq_coe_iff, FirstOrder.Language.ElementaryEmbedding.infinite_realizations_iff}
\leanok
\uses{def:elementary-maps}
Let \(e:M\to N\) be an elementary embedding and let \(X_M\) and \(X_N\)
be the realization sets of the same formula with parameters transported
along \(e\).  Then \(e\) induces an embedding \(X_M\hookrightarrow X_N\).
Moreover \(X_M\) is finite if and only if \(X_N\) is finite, and in the
finite case they have the same cardinality.
\end{lemma}

\begin{proof}
\leanok
\uses{def:counting-formulas}
Elementarity preserves the counting formulas, while coordinatewise
application of \(e\) gives the cardinal inequality.
\end{proof}

\begin{theorem}[Compactness]
\label{thm:compactness}
\lean{FirstOrder.Language.Theory.isSatisfiable_iff_isFinitelySatisfiable}
\mathlibok
A theory is satisfiable if and only if every finite subset of it is
satisfiable.
\end{theorem}

\begin{theorem}[Downward Loewenheim--Skolem]
\label{thm:downward-lowenheim-skolem}
\lean{FirstOrder.Language.exists_elementarySubstructure_card_eq}
\mathlibok
\uses{def:elementary-maps}
Let \(M\) be an \(L\)-structure and \(X\subseteq M\).  Then \(X\) is
contained in an elementary submodel \(N\preceq M\) whose cardinality is
bounded by \(|X|+|L|+\aleph_0\).
\end{theorem}

\begin{definition}[Elementary chain]
\label{def:elementary-chain}
\lean{FirstOrder.Language.ElementaryChain}
\leanok
\uses{def:elementary-maps}
An elementary chain is an increasing family \((M_i)_{i<\alpha}\) with
\(M_i\preceq M_j\) for \(i\leq j\), taking unions at limit stages.
\end{definition}

\begin{theorem}[Tarski's elementary chain theorem]
\label{thm:elementary-chain-union}
\lean{FirstOrder.Language.ElementaryChain.Limit, FirstOrder.Language.ElementaryChain.toLimitElementary, FirstOrder.Language.ElementaryChain.limit_models}
\leanok
\uses{def:elementary-chain}
The union of an elementary chain carries a natural \(L\)-structure, is a
model of every theory modeled by the stages, and every stage embeds
elementarily into the union.
\end{theorem}

\begin{proof}
\uses{thm:tarski-vaught-test}
Apply Tarski--Vaught: every finite tuple of parameters and every witness
appears at some common stage.
\end{proof}

\begin{lemma}[Cardinality of elementary chain unions]
\label{lem:elementary-chain-cardinality}
\lean{FirstOrder.Language.ElementaryChain.mk_limit_le_of_lift_mk_le, FirstOrder.Language.ElementaryChain.mk_limit_eq_of_forall_lift_mk_eq}
\leanok
\uses{def:elementary-chain}
Let \((M_i)_{i<\alpha}\) be an elementary chain.  If \(\kappa\) is
infinite, each \(|M_i|\leq\kappa\), and \(|\alpha|\leq\kappa\), then the
union has cardinality at most \(\kappa\).  If every stage has cardinality
\(\kappa\), then the union has cardinality exactly \(\kappa\).
\end{lemma}

\begin{proof}
The union is a surjective image of the disjoint union of the stages.
\end{proof}

\begin{theorem}[Elementary-chain construction principle]
\label{thm:loewenheim-skolem-constructions}
\lean{FirstOrder.Language.ElementaryChain.toLimitElementary, FirstOrder.Language.ElementaryChain.limit_models, FirstOrder.Language.ElementaryChain.mk_limit_le_of_lift_mk_le, FirstOrder.Language.ElementaryChain.mk_limit_eq_of_forall_lift_mk_eq}
\leanok
\uses{def:elementary-chain}
Any inductive construction by elementary extensions, with cardinality
bounds at successor stages, yields controlled limit models by taking the
elementary-chain limit.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:elementary-chain-union,lem:elementary-chain-cardinality}
Combine the elementary-chain theorem with the cardinality bound.
\end{proof}

\section{Types, Isolation, and Prime Constructions}

\begin{definition}[Types over parameters]
\label{def:types-over-parameters}
\lean{FirstOrder.Language.CompleteTypeOver}
\leanok
\uses{def:elementary-maps}
For \(A\subseteq M\models T\), an \(n\)-type over \(A\) is a set of
\(L_A\)-formulas in \(n\) free variables consistent with
\(\operatorname{Th}_A(M)\).  A complete type decides every such formula;
write \(S_n^M(A)\) for the space of complete \(n\)-types over \(A\).
\end{definition}

\begin{definition}[Realization, omission, and isolation]
\label{def:isolated-and-omitted-types}
\lean{FirstOrder.Language.Theory.CompleteType.RealizedBy, FirstOrder.Language.Theory.CompleteType.IsRealizedIn, FirstOrder.Language.Theory.CompleteType.IsOmittedIn, FirstOrder.Language.Theory.CompleteType.IsIsolated, FirstOrder.Language.Theory.CompleteType.IsolatedBy}
\leanok
\uses{def:types-over-parameters}
A tuple realizes a type if it satisfies every formula in the type.  A
model omits a type if no tuple realizes it.  A complete type is isolated
if some formula in it determines that type uniquely.
\end{definition}

\begin{definition}[Omega-stability]
\label{def:omega-stability}
\lean{FirstOrder.Language.Theory.IsOmegaStable}
\leanok
\uses{def:types-over-parameters}
The theory \(T\) is \(\omega\)-stable if every finite-arity type space
over every countable parameter set is countable.
\end{definition}

\begin{lemma}[Splitting a non-isolated basic open set]
\label{lem:isolated-splitting}
\lean{FirstOrder.Language.Theory.CompleteType.exists_isolated_splitting}
\leanok
\uses{def:isolated-and-omitted-types}
If a nonempty basic open set in a type space contains no isolated type,
then it can be split by a formula into two nonempty basic open subsets,
neither of which contains an isolated type.
\end{lemma}

\begin{proof}
\leanok
Otherwise the original basic open set would determine a unique complete
type and isolate it.
\end{proof}

\begin{lemma}[Projection of an isolated type]
\label{lem:isolated-projection}
\lean{FirstOrder.Language.Theory.CompleteType.isIsolated_typeOf_left}
\leanok
\uses{def:isolated-and-omitted-types}
If \((\bar a,\bar b)\) realizes an isolated type over \(A\), then
\(\bar a\) realizes an isolated type over \(A\).
\end{lemma}

\begin{proof}
\leanok
Existentially quantify the \(\bar b\)-variables in an isolating formula.
\end{proof}

\begin{lemma}[Transitivity of isolation]
\label{lem:isolated-transitivity}
\lean{FirstOrder.Language.Theory.CompleteType.isIsolated_typeOf_trans}
\leanok
\uses{def:isolated-and-omitted-types}
Let \(A\subseteq B\subseteq M\).  If every finite tuple from \(B\)
realizes an isolated type over \(A\), and \(\bar a\) realizes an isolated
type over \(B\), then \(\bar a\) realizes an isolated type over \(A\).
\end{lemma}

\begin{proof}
\leanok
\uses{lem:isolated-projection}
Combine the isolating formula over \(B\) with an isolating formula over
\(A\) for the finitely many parameters it uses, then project.
\end{proof}

\begin{lemma}[Binary splitting obstruction]
\label{lem:binary-splitting-obstruction}
\lean{FirstOrder.Language.Theory.IsOmegaStable.false_of_formula_splitting}
\leanok
\uses{def:omega-stability}
In an \(\omega\)-stable theory there is no realizable property of
formulas that starts at a root and can be preserved under both sides of
a binary split at every node.
\end{lemma}

\begin{proof}
\leanok
A full binary tree of such formulas uses only countably many parameters,
and compactness yields distinct complete types along its
\(2^{\aleph_0}\) branches.
\end{proof}

\begin{theorem}[Density of isolated types in omega-stable theories]
\label{thm:isolated-types-dense}
\lean{FirstOrder.Language.Theory.CompleteType.exists_isolated_mem_typesWith}
\leanok
\uses{def:omega-stability,def:isolated-and-omitted-types}
If \(T\) is \(\omega\)-stable, isolated types are dense in every
finite-arity type space over parameters.
\end{theorem}

\begin{proof}
\leanok
\uses{lem:isolated-splitting,lem:binary-splitting-obstruction}
Otherwise repeatedly split a nonempty basic open set containing no
isolated type.
\end{proof}

\begin{lemma}[Type spaces and partial elementary maps]
\label{lem:type-space-homeomorphism}
\lean{FirstOrder.Language.PartialElementaryEmbedding.completeTypeOverHomeomorph}
\leanok
\uses{def:types-over-parameters,def:partial-elementary-embedding}
A partial elementary map \(f:A\to B\) induces a homeomorphism between
the complete type spaces over \(A\) and \(B\).
\end{lemma}

\begin{proof}
\leanok
\uses{thm:compactness}
Transport parameters along \(f\); partial elementarity preserves finite
satisfiability, and compactness produces the transported complete type.
\end{proof}

\begin{definition}[Prime models over parameter sets]
\label{def:prime-over}
\uses{def:elementary-maps,def:partial-elementary-embedding}
A model is prime over \(A\) if every partial elementary map from \(A\)
into a model of \(T\) extends to an elementary embedding of the whole
model.
\end{definition}

\begin{theorem}[Prime extension over a set]
\label{thm:prime-extension-over-set}
\uses{def:omega-stability,def:prime-over,def:isolated-and-omitted-types}
Let \(T\) be \(\omega\)-stable, \(M\models T\), and \(A\subseteq M\).
There is an elementary submodel \(M_0\preceq M\) prime over \(A\), and
\(M_0\) may be chosen so that every finite tuple from \(M_0\) realizes
an isolated type over \(A\).
\end{theorem}

\begin{proof}
\uses{thm:tarski-vaught-test,thm:isolated-types-dense,lem:type-space-homeomorphism,lem:isolated-transitivity}
Construct the model by adjoining witnesses whose types over the current
stage are isolated.  Density supplies the witnesses, Tarski--Vaught
proves elementarity, and the construction order lets any partial
elementary map on \(A\) extend through the construction.  Isolation
transitivity records atomicity over the original base.
\end{proof}

\begin{lemma}[Isomorphisms preserve definable set cardinalities]
\label{lem:isomorphism-preserves-definable-set-cardinality}
\uses{def:elementary-maps}
An isomorphism restricts to a bijection between corresponding
parameter-definable sets.
\end{lemma}

\begin{proof}
Use elementarity in both directions and restrict the isomorphism.
\end{proof}

\begin{lemma}[Isomorphisms preserve realized type counts]
\label{lem:isomorphism-preserves-realized-type-counts}
\uses{def:types-over-parameters,def:elementary-maps}
If \(f:M\cong N\) and \(A\subseteq M\), then \(f\) induces a bijection
between the complete types over \(A\) realized in \(M\) and the complete
types over \(f(A)\) realized in \(N\).
\end{lemma}

\begin{proof}
Send the type of \(\bar a\) over \(A\) to the type of \(f\bar a\) over
\(f(A)\), and use \(f^{-1}\) for the inverse.
\end{proof}

\section{Categoricity Implies Omega-Stability}

\begin{theorem}[Skolem expansion]
\label{thm:skolem-expansion}
For every \(L\)-theory \(T\), there is an expanded language \(L^*\) and
a theory \(T^*\supseteq T\) with Skolem functions such that every model
of \(T\) admits an expansion to a model of \(T^*\), every model of
\(T^*\) has an \(L\)-reduct satisfying \(T\), and
\(|L^*|=|L|+\aleph_0\) may be arranged.
\end{theorem}

\begin{theorem}[Infinite Ramsey theorem]
\label{thm:infinite-ramsey}
For every finite coloring of the \(n\)-element subsets of a countably
infinite set, there is an infinite homogeneous subset.
\end{theorem}

\begin{definition}[Order indiscernibles]
\label{def:order-indiscernibles}
Let \((I,<)\) be a linear order.  An injective sequence
\((x_i)_{i\in I}\) in a model is a sequence of order indiscernibles if
any two increasing finite subsequences of the same length satisfy the
same formulas.
\end{definition}

\begin{theorem}[Existence of order indiscernibles]
\label{thm:existence-of-order-indiscernibles}
\uses{def:order-indiscernibles}
If \(T\) has infinite models and \(I\) is an infinite linear order, then
some model of \(T\) contains an injective sequence of order
indiscernibles indexed by \(I\).
\end{theorem}

\begin{proof}
\uses{thm:compactness,thm:infinite-ramsey}
For each finite fragment of the Ehrenfeucht--Mostowski diagram, use
Ramsey's theorem inside an infinite model to find a sufficiently
homogeneous finite sequence.  Compactness realizes all prescribed
finite ordered patterns simultaneously.
\end{proof}

\begin{definition}[Skolem hull]
\label{def:skolem-hull}
\uses{thm:skolem-expansion}
If \(M\models T^*\) and \(X\subseteq M\), the \(L^*\)-substructure
generated by \(X\) is the Skolem hull \(H(X)\).
\end{definition}

\begin{theorem}[Skolem hull reducts are elementary]
\label{thm:skolem-hull-elementary}
\uses{def:skolem-hull}
If \(M\models T^*\), then the reduct \(H(X)\!\upharpoonright_L\) is an
elementary substructure of \(M\!\upharpoonright_L\); in particular it is
a model of \(T\).
\end{theorem}

\begin{proof}
\uses{thm:tarski-vaught-test}
For an existential \(L\)-formula with parameters in the hull, the
Skolem function added for that \(L\)-formula supplies a witness that is
again in the hull.  Apply Tarski--Vaught to the reducts.
\end{proof}

\begin{lemma}[Few relative order types over a well-order]
\label{lem:few-relative-order-types}
Let \(I\) be a well-order, let \(J\subseteq I\), and let \(r<\omega\).
Then the quantifier-free order types over \(J\) of \(r\)-tuples from
\(I\) form a set of cardinality at most \(|J|+\aleph_0\).
\end{lemma}

\begin{proof}
For one coordinate, its order relations to \(J\) are determined by the
initial segment of \(J\) below it together with whether it equals an
element of \(J\).  The inherited well-order on \(J\) has at most
\(|J|+1\) initial segments.  For a fixed finite tuple, also record the
finitely many order and equality relations among its coordinates; finite
products preserve the bound \(|J|+\aleph_0\).
\end{proof}

\begin{theorem}[Models realizing few types]
\label{thm:few-types-model}
\uses{def:types-over-parameters}
Let \(L\) be countable and \(T\) have infinite models.  For every
\(\kappa\geq\aleph_0\), there is a model \(M\models T\) of cardinality
\(\kappa\) such that for every \(A\subseteq M\) and finite \(n\geq1\),
\(M\) realizes at most \(|A|+\aleph_0\) complete \(n\)-types over \(A\).
\end{theorem}

\begin{proof}
\uses{thm:skolem-expansion,thm:existence-of-order-indiscernibles,thm:skolem-hull-elementary,lem:few-relative-order-types}
Take a countable Skolem expansion, build order indiscernibles in the
expanded language \(L^*\) indexed by the initial ordinal of size
\(\kappa\), and take their Skolem hull.  The hull has size exactly
\(\kappa\).  A parameter set \(A\) involves only \(|A|+\aleph_0\)
indices in its Skolem terms; call their set \(I_A\).  For any fixed
finite tuple of Skolem terms, \(L^*\)-order indiscernibility shows that
the complete type of its value over \(A\) is determined by the relative
order type over \(I_A\) of the finitely many indices occurring in the
terms.  Because the index order is a well-order, the preceding lemma
bounds the number of such relative order types by \(|A|+\aleph_0\).
There are only countably many finite tuples of Skolem terms, so the same
bound holds for the realized finite-arity types in the reduct.
\end{proof}

\begin{lemma}[Non-omega-stability produces a large model realizing many types]
\label{lem:non-omega-stable-large-model-realizing-many-types}
\uses{def:omega-stability,def:types-over-parameters}
If \(T\) is not \(\omega\)-stable and \(\kappa\geq\aleph_1\), then
there is a model of cardinality \(\kappa\) realizing uncountably many
complete types over some countable parameter set.
\end{lemma}

\begin{proof}
\uses{thm:compactness,thm:downward-lowenheim-skolem,thm:loewenheim-skolem-constructions}
Choose uncountably many types over a countable set, add constants
requesting realizations of \(\aleph_1\) of them, use compactness, and
then adjust the model to cardinality \(\kappa\) by Loewenheim--Skolem
and elementary-chain constructions.
\end{proof}

\begin{theorem}[Uncountable categoricity implies omega-stability]
\label{thm:uncountable-categoricity-implies-omega-stable}
\uses{def:categoricity,def:omega-stability}
If \(T\) is \(\kappa\)-categorical for some uncountable \(\kappa\), then
\(T\) is \(\omega\)-stable.
\end{theorem}

\begin{proof}
\uses{lem:non-omega-stable-large-model-realizing-many-types,thm:few-types-model,lem:isomorphism-preserves-realized-type-counts}
Otherwise one model of size \(\kappa\) realizes uncountably many types
over a countable set while a few-type model of the same size realizes
only countably many over every countable set, contradicting categoricity.
\end{proof}

\section{Vaughtian Pairs and the Second Forward Milestone}

\subsection{Definably full comparison models}

\begin{lemma}[One-step enlargement of parameter-definable sets]
\label{lem:one-step-full-definable-sets}
\lean{FirstOrder.Language.Theory.exists_elementaryExtension_card_eq_with_full_unary_realizations}
\leanok
\uses{def:elementary-maps}
Let \(M\models T\) have infinite cardinality \(\kappa\), with
\(|L|\leq\kappa\).  There are a model \(N\models T\) of cardinality
\(\kappa\) and an elementary embedding \(e:M\to N\) such that every
infinite unary set definable with parameters from \(M\) has
\(\kappa\) realizations in \(N\) after transporting its parameters.
\end{lemma}

\begin{proof}
\leanok
\uses{thm:compactness,thm:downward-lowenheim-skolem}
Add the elementary diagram of \(M\), together with \(\kappa\) distinct
constants in each infinite unary definable set.  Compactness gives an
elementary extension and Loewenheim--Skolem returns to size \(\kappa\).
\end{proof}

\begin{proposition}[Large models with no small infinite definable sets]
\label{prop:large-model-no-small-infinite-definable-sets}
\lean{FirstOrder.Language.Theory.exists_model_of_card_definably_full}
\leanok
Let \(T\) be in a countable language with infinite models and let
\(\kappa\geq\aleph_1\).  Then \(T\) has a model of cardinality
\(\kappa\) in which every parameter-definable unary set is finite or has
cardinality \(\kappa\).
\end{proposition}

\begin{proof}
\leanok
\uses{lem:one-step-full-definable-sets,lem:elementary-embedding-definable-set-cardinality,thm:loewenheim-skolem-constructions}
Iterate the one-step enlargement along a countable elementary chain.
Every finite parameter tuple appears at a stage; if its definable set is
infinite in the limit, elementarity shows that it was already infinite
at that stage and the next stage gives it \(\kappa\) realizations.
\end{proof}

\subsection{Two-cardinal models and Vaughtian pairs}

\begin{definition}[\((\kappa,\lambda)\)-models]
\label{def:kappa-lambda-model}
For \(\kappa>\lambda\geq\aleph_0\), a \((\kappa,\lambda)\)-model is a
model of size \(\kappa\) with a one-variable parameter-definable subset
of size \(\lambda\).
\end{definition}

\begin{definition}[Vaughtian pairs]
\label{def:vaughtian-pair}
\lean{FirstOrder.Language.Theory.IsVaughtianPair, FirstOrder.Language.Theory.HasVaughtianPair}
\leanok
\uses{def:elementary-maps}
A Vaughtian pair is a proper elementary pair \(M\preceq N\) together
with an infinite unary set definable over \(M\) whose realization set
does not grow in \(N\).
\emph{Formalization note.}  Lean presents the pair by an elementary
embedding \(e:M\hookrightarrow N\), properness by non-surjectivity, and
unchanged realizations by equality with the image of the realization set
in \(M\).
\end{definition}

\begin{lemma}[Embedding and inclusion presentations of Vaughtian pairs]
\label{lem:vaughtian-pair-inclusion-presentation}
\uses{def:vaughtian-pair,def:elementary-maps}
Every Vaughtian pair presented by a non-surjective elementary embedding
is isomorphic to one presented by a literal inclusion \(M'\preceq N\),
with the witnessing formula and realization set transported to the image
substructure.
\end{lemma}

\begin{proof}
Replace the source model by the image substructure of the elementary
embedding.  The embedding is an isomorphism onto that image, and the
Vaughtian witness transports along this isomorphism.
\end{proof}

\begin{lemma}[\((\kappa,\lambda)\)-models give Vaughtian pairs]
\label{lem:kappa-lambda-model-gives-vaughtian-pair}
\uses{def:kappa-lambda-model,def:vaughtian-pair}
If \(T\) has a \((\kappa,\lambda)\)-model with
\(\kappa>\lambda\geq\aleph_0\), then \(T\) has a Vaughtian pair.
\end{lemma}

\begin{proof}
\uses{thm:downward-lowenheim-skolem}
Take an elementary submodel of size \(\lambda\) containing the whole
small definable set and its parameters.
\end{proof}

\begin{corollary}[No small infinite definable sets without Vaughtian pairs]
\label{cor:no-small-infinite-definable-sets-without-vaughtian-pairs}
\uses{def:vaughtian-pair}
If \(T\) has no Vaughtian pairs and \(M\models T\) has size \(\kappa\),
then every infinite parameter-definable unary subset of \(M\) has size
\(\kappa\).
\end{corollary}

\begin{proof}
\uses{lem:kappa-lambda-model-gives-vaughtian-pair}
Any intermediate infinite cardinality would give a two-cardinal model
and hence a Vaughtian pair.
\end{proof}

\subsection{Pair language and homogeneous copies}

\begin{definition}[The pair language]
\label{def:pair-language}
\uses{def:elementary-maps}
For an inclusion \(M\preceq N\), let \(L_0=L\cup\{U\}\), where the new
unary predicate \(U\) is interpreted as \(M\).  For each \(L\)-formula
\(\varphi\), its relativization \(\varphi^U\) restricts all quantifiers
to \(U\).
\end{definition}

\begin{lemma}[Relativization in an elementary pair]
\label{lem:pair-relativization}
\uses{def:pair-language}
If \(M\preceq N\), then for every \(L\)-formula \(\varphi\) and tuple
\(\bar a\) from \(M\), the pair structure \((N,M)\) satisfies
\(\varphi^U(\bar a)\) if and only if \(M\models\varphi(\bar a)\).
\end{lemma}

\begin{proof}
Induct on formulas.  Atomic formulas are unchanged, Boolean connectives
are immediate, and the quantified step uses that \(U\) is interpreted
as exactly \(M\).
\end{proof}

\begin{lemma}[Elementary pairs are first-order and preserved]
\label{lem:elementary-pair-axiomatization}
\uses{def:pair-language,lem:pair-relativization,thm:tarski-vaught-test}
There is an \(L_0\)-theory whose models are exactly the structures
\((N,U)\) in which \(N\models T\), the predicate \(U\) is the universe
of an \(L\)-substructure \(M\subseteq N\), and \(M\preceq N\).
Consequently, an \(L_0\)-elementary substructure or elementary extension
of an elementary pair is again an elementary pair on its \(U\)-part and
ambient \(L\)-reduct.
\end{lemma}

\begin{proof}
Include \(T\), axioms saying that \(U\) contains the constants and is
closed under the \(L\)-functions, and for every \(L\)-formula
\(\varphi(\bar x)\) the scheme
\[
  \forall \bar x\in U\,
  \bigl(\varphi(\bar x)\leftrightarrow\varphi^U(\bar x)\bigr).
\]
A genuine elementary pair satisfies this scheme by relativization.
Conversely, the closure axioms make the \(U\)-part a substructure and
the scheme gives the Tarski--Vaught condition, hence elementarity.
Since this is a first-order theory, elementary substructures and
extensions of pair structures remain models of the same theory.
\end{proof}

\begin{definition}[Homogeneous models]
\label{def:homogeneous-model}
\uses{def:partial-elementary-embedding}
For infinite \(\kappa\), a model is \(\kappa\)-homogeneous if every
partial elementary map with domain of size below \(\kappa\) can be
extended over any one additional element.  A model is homogeneous if it
is \(|M|\)-homogeneous.
\end{definition}

\begin{theorem}[Back-and-forth for countable homogeneous models]
\label{thm:homogeneous-back-and-forth}
\uses{def:homogeneous-model,def:types-over-parameters}
If \(M\) and \(N\) are countable homogeneous models realizing the same
complete finite-arity types over \(\emptyset\), then every finite
partial elementary map between them extends to an isomorphism.
\end{theorem}

\begin{proof}
Enumerate both models and alternate forth and back extension steps.
Homogeneity moves a realization of the relevant joint type so that it
extends the current partial elementary map.
\end{proof}

\begin{lemma}[Countable Vaughtian pair]
\label{lem:countable-vaughtian-pair}
\uses{def:vaughtian-pair,def:pair-language}
If \(T\) is in a countable language and has a Vaughtian pair, then it
has a Vaughtian pair \((N_0,M_0)\) with \(N_0\) countable.
\end{lemma}

\begin{proof}
\uses{lem:vaughtian-pair-inclusion-presentation,lem:pair-relativization,lem:elementary-pair-axiomatization,thm:downward-lowenheim-skolem}
Pass to the inclusion presentation and apply downward
Loewenheim--Skolem in the pair language, retaining the parameters of the
Vaughtian witness.  The pair axioms ensure that the resulting
\(U\)-part is still elementary in the ambient reduct, while pair
elementarity preserves properness, infinitude, and the assertion that no
witness lies outside the \(U\)-part.
\end{proof}

\begin{lemma}[Realizing a type on the submodel side]
\label{lem:realize-type-in-submodel-side}
\uses{def:pair-language,def:isolated-and-omitted-types}
Let \(M_0\preceq N_0\) be countable.  If a type over parameters from
\(M_0\) is realized in \(N_0\), then some countable elementary extension
of the pair realizes that type on the \(M\)-side.
\end{lemma}

\begin{proof}
\uses{lem:pair-relativization,lem:elementary-pair-axiomatization,thm:compactness,thm:downward-lowenheim-skolem}
Relativize the type to \(U\), add it to the elementary diagram of the
pair, and use compactness followed by downward Loewenheim--Skolem.  The
pair axioms ensure that the resulting structure is again an elementary
pair.
\end{proof}

\begin{lemma}[Realizing a type in an elementary extension of pairs]
\label{lem:realize-type-in-extension-pair}
\uses{def:pair-language,def:isolated-and-omitted-types}
Any type over parameters from the larger side of a countable elementary
pair is realized in some countable elementary extension of that pair.
\end{lemma}

\begin{proof}
\uses{lem:elementary-pair-axiomatization,thm:compactness,thm:downward-lowenheim-skolem}
Add the type to the elementary diagram of the pair and apply compactness
and downward Loewenheim--Skolem; the pair axioms preserve the elementary
pair structure.
\end{proof}

\begin{lemma}[Homogeneous countable Vaughtian pair]
\label{lem:homogeneous-vaughtian-pair}
\uses{def:vaughtian-pair,def:pair-language,def:homogeneous-model}
Every countable elementary pair has a countable elementary extension
pair whose two sides are homogeneous and realize the same complete
finite-arity types over \(\emptyset\).  Applied to a Vaughtian pair, the
Vaughtian witness persists.
\end{lemma}

\begin{proof}
\uses{lem:realize-type-in-submodel-side,lem:realize-type-in-extension-pair,thm:loewenheim-skolem-constructions}
Iterate the requirements that types seen on the larger side appear on
the smaller side and that every one-point homogeneity request is
realized.  A countable elementary-chain limit satisfies all requests.
\end{proof}

\begin{lemma}[Copying a Vaughtian successor over its parameters]
\label{lem:vaughtian-pair-copy-over-parameters}
\uses{def:vaughtian-pair,thm:homogeneous-back-and-forth}
Let \((N,M)\) be a countable homogeneous Vaughtian pair whose two sides
realize the same finite-arity types, witnessed by parameters \(\bar a\)
from \(M\).  Let \(P\) be another countable homogeneous model realizing
the same types, let \(\bar a_P\) be a tuple from \(P\), and suppose the
map \(\bar a\mapsto\bar a_P\) is partial elementary from \(M\) to \(P\).
Then there is an isomorphism \(h:M\cong P\) carrying \(\bar a\) to
\(\bar a_P\).  Transporting \(M\preceq N\) along \(h\) gives a proper
elementary extension of \(P\) adding no new realization of the
transported Vaughtian witness.
\end{lemma}

\begin{proof}
Extend the specified finite partial elementary map by homogeneous
back-and-forth, then transport the larger side of the pair across the
resulting isomorphism.
\end{proof}

\begin{theorem}[Vaughtian pairs produce an \((\aleph_1,\aleph_0)\)-model]
\label{thm:vaughtian-pair-gives-aleph-one-aleph-zero-model}
\uses{def:vaughtian-pair,def:kappa-lambda-model}
If \(T\) is in a countable language and has a Vaughtian pair, then it
has an \((\aleph_1,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:countable-vaughtian-pair,lem:homogeneous-vaughtian-pair,lem:vaughtian-pair-copy-over-parameters,thm:loewenheim-skolem-constructions}
Start with a countable homogeneous Vaughtian pair and fix its witnessing
parameters.  Build an elementary chain of length \(\omega_1\), using the
copying lemma at each successor stage.  There the distinguished tuple is
already contained in the current model, so use that same tuple on both
sides of the finite partial elementary map; the copied successor
therefore preserves the fixed witness.  At a countable limit stage the
union is countable; every finite tuple and every finite partial
elementary map is contained in some earlier stage, so homogeneity is
inherited, and every finite tuple realizes one of the same
\(\emptyset\)-types as the fixed pair.  Thus the limit again satisfies
the hypotheses of the copying lemma.  Every successor is proper, so the
final union has size \(\aleph_1\), while the witnessing definable set
never grows and remains countably infinite.
\end{proof}

\begin{theorem}[Vaught's two-cardinal theorem]
\label{thm:vaught-two-cardinal}
\uses{def:kappa-lambda-model}
If \(T\) is in a countable language and has a \((\kappa,\lambda)\)-model
for some \(\kappa>\lambda\geq\aleph_0\), then \(T\) has an
\((\aleph_1,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:kappa-lambda-model-gives-vaughtian-pair,thm:vaughtian-pair-gives-aleph-one-aleph-zero-model}
Pass from the two-cardinal model to a Vaughtian pair and then to an
\((\aleph_1,\aleph_0)\)-model.
\end{proof}

\subsection{Omega-stable upward transfer}

\begin{lemma}[Large definable set with no uncountable definable split]
\label{lem:omega-stable-large-definable-almost-minimal}
\uses{def:omega-stability}
If \(T\) is \(\omega\)-stable and \(|M|\geq\aleph_1\), then some
parameter-definable unary set \(D\subseteq M\) is uncountable and every
subset of \(D\) definable with parameters from \(M\) is countable or
co-countable in \(D\).
\end{lemma}

\begin{proof}
\uses{lem:binary-splitting-obstruction}
Otherwise recursively split every uncountable \(M\)-parameter-definable
set into two uncountable \(M\)-parameter-definable subsets.
\end{proof}

\begin{lemma}[The co-countable generic is realized outside the base]
\label{lem:cocountable-generic-type}
Let \(D\subseteq M\) be uncountable and definable such that every subset
of \(D\) definable with parameters from \(M\) is countable or
co-countable.  The partial type over \(M\) consisting of the formulas
whose traces on \(D\) are co-countable is finitely satisfiable in
\(M\), and it has a realization \(c\) in an elementary extension of
\(M\) with \(c\notin M\).
\end{lemma}

\begin{proof}
\uses{thm:compactness}
A finite intersection of co-countable subsets of an uncountable set is
nonempty, so compactness realizes the partial type in an elementary
extension.  The type contains the formula defining \(D\), while for
each \(m\in D(M)\) it contains \(x\neq m\), since
\(D\setminus\{m\}\) is co-countable in \(D\).  Hence no element of
\(M\) realizes the type.
\end{proof}

\begin{lemma}[Specializing isolated types along a co-countable generic]
\label{lem:cocountable-generic-specialization}
\uses{def:isolated-and-omitted-types}
Let \(D\subseteq M\) be uncountable and definable, with every subset of
\(D\) definable with parameters from \(M\) countable or co-countable.
Let \(M\preceq N\), and let \(c\in N\) satisfy every formula over
\(M\) whose trace on \(D\) is co-countable.  If a finite tuple
\(\bar b\in N\) realizes an isolated type over \(M\cup\{c\}\), then
every countable partial type over \(M\) realized by \(\bar b\) is
already realized in \(M\).
\end{lemma}

\begin{proof}
Let \(\theta(\bar x,c,\bar m)\) isolate the type of \(\bar b\) over
\(M\cup\{c\}\), and enumerate a countable partial type realized by
\(\bar b\) as \(\{\psi_i(\bar x):i<\omega\}\), allowing parameters
from \(M\) in the \(\psi_i\).  For each \(i\), isolation makes
\[
  E_i(y):=\forall\bar x\,
  (\theta(\bar x,y,\bar m)\to\psi_i(\bar x))
\]
true of \(c\); likewise
\(E_*(y):=\exists\bar x\,\theta(\bar x,y,\bar m)\) is true of \(c\).
Any formula over \(M\) true of \(c\) must have co-countable trace on
\(D\): otherwise its complement would be co-countable and would also
belong to the generic realized by \(c\).  Thus every \(E_i\) and
\(E_*\) has co-countable trace.  The intersection of countably many
co-countable subsets of the uncountable set \(D\) is nonempty.  Choose
\(d\in D(M)\) in that intersection.  Then \(M\models E_*(d)\), so a
tuple in \(M\) realizes \(\theta(\bar x,d,\bar m)\), and the formulas
\(E_i(d)\) force that tuple to realize every \(\psi_i\).
\end{proof}

\begin{lemma}[Proper extension realizing no new countable finite-arity types]
\label{lem:proper-extension-realizes-no-new-countable-types}
\uses{def:omega-stability,def:isolated-and-omitted-types}
Suppose \(T\) is \(\omega\)-stable and \(|M|\geq\aleph_1\).  Then there
is a proper elementary extension \(N\succeq M\) of the same cardinality
such that every countable finite-arity partial type over \(M\) realized
in \(N\) is already realized in \(M\).
\end{lemma}

\begin{proof}
\uses{lem:omega-stable-large-definable-almost-minimal,lem:cocountable-generic-type,lem:cocountable-generic-specialization,thm:prime-extension-over-set,thm:downward-lowenheim-skolem}
Choose \(D\) by the almost-minimal definable-set lemma and realize its
co-countable generic by \(c\notin M\).  Downward Loewenheim--Skolem gives
an elementary extension of cardinality \(|M|\) containing
\(M\cup\{c\}\).  Inside it take the prime extension over
\(M\cup\{c\}\), chosen so that every finite tuple realizes an isolated
type over that base.  This model still has size \(|M|\) and is proper
over \(M\) because it contains \(c\).  For any countable finite-arity
partial type over \(M\) realized there, apply the specialization lemma
to the realizing tuple to obtain a realization already in \(M\).
\end{proof}

\begin{lemma}[No-new-countable-types extensions preserve countable definable sets]
\label{lem:no-new-countable-types-preserve-countable-definable-sets}
Let \(M\preceq N\), and suppose every countable finite-arity partial type
over \(M\) realized in \(N\) is already realized in \(M\).  If
\(D=\varphi(M,\bar a)\) is countable for parameters \(\bar a\in M\),
then
\[
  \varphi(N,\bar a)=D.
\]
\end{lemma}

\begin{proof}
If \(c\in\varphi(N,\bar a)\setminus M\), then \(c\) realizes the
countable partial type over \(M\)
\[
  \{\varphi(x,\bar a)\}\cup\{x\neq d:d\in D\}.
\]
By hypothesis this type is realized by some element of \(M\), but every
point of \(M\) satisfying \(\varphi(x,\bar a)\) already lies in \(D\),
a contradiction.  Elementarity gives the reverse inclusion.
\end{proof}

\begin{theorem}[Two-cardinal transfer above \(\aleph_1\)]
\label{thm:two-cardinal-transfer}
\uses{def:omega-stability,def:kappa-lambda-model}
Suppose \(T\) is \(\omega\)-stable and has an
\((\aleph_1,\aleph_0)\)-model.  If \(\kappa>\aleph_1\), then \(T\) has
a \((\kappa,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:proper-extension-realizes-no-new-countable-types,lem:no-new-countable-types-preserve-countable-definable-sets,thm:loewenheim-skolem-constructions}
Iterate proper same-cardinality elementary extensions that realize no new
countable partial types.  At each successor stage the preceding lemma
shows that the countable definable set from the initial model gains no
new realizations at all.  At limit stages its realization set is the
union of the unchanged realization sets, so it remains the original
countable set.  Proper successor stages force the final union to have
size \(\kappa\).
\end{proof}

\begin{theorem}[Uncountable categoricity rules out Vaughtian pairs]
\label{thm:uncountable-categoricity-no-vaughtian-pairs}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Suppose \(T\) is \(\omega\)-stable and \(\kappa\)-categorical for some
uncountable \(\kappa\).  Then \(T\) has no Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:vaughtian-pair-gives-aleph-one-aleph-zero-model,thm:two-cardinal-transfer,prop:large-model-no-small-infinite-definable-sets,lem:isomorphism-preserves-definable-set-cardinality}
A Vaughtian pair gives a model of size \(\kappa\) with an infinite
countable definable set (directly if \(\kappa=\aleph_1\), by transfer
otherwise).  The definably-full construction gives another model of
size \(\kappa\) in which every infinite definable set has size
\(\kappa\), contradicting categoricity.
\end{proof}

\begin{theorem}[Uncountable categoricity implies omega-stability and no Vaughtian pairs]
\label{thm:uncountable-categoricity-implies-omega-stable-no-vaughtian-pairs}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
If \(T\) is categorical in some uncountable cardinal, then it is
\(\omega\)-stable and has no Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:uncountable-categoricity-implies-omega-stable,thm:uncountable-categoricity-no-vaughtian-pairs}
Apply the two forward milestones in order.
\end{proof}

\section{Strong Minimality and Dimension}

\begin{definition}[Minimal and strongly minimal formulas]
\label{def:minimality}
Let \(D\) be an infinite unary definable set.  It is minimal in \(M\)
if every definable subset of \(D\) is finite or cofinite in \(D\), and
strongly minimal if this remains true in every elementary extension.
\end{definition}

\begin{definition}[Algebraic closure on a strongly minimal set over a base]
\label{def:strongly-minimal-pregeometry}
\uses{def:minimality}
For a strongly minimal set \(D\) definable over \(A\), define
\[
  \operatorname{cl}_A(X)=D\cap\operatorname{acl}(A\cup X).
\]
Thus \(b\in\operatorname{cl}_A(X)\) exactly when \(b\in D\) is
algebraic over \(A\cup X\).
\end{definition}

\begin{lemma}[Closure laws for relative algebraic closure]
\label{lem:strongly-minimal-closure-laws}
\uses{def:strongly-minimal-pregeometry}
The operator \(\operatorname{cl}_A\) is reflexive, monotone, idempotent,
and has finite character.
\end{lemma}

\begin{proof}
These are the corresponding standard properties of algebraic closure,
restricted to \(D\) after adjoining the fixed base \(A\).
\end{proof}

\begin{lemma}[Nonalgebraic types have arbitrarily many distinct conjugates]
\label{lem:nonalgebraic-type-many-realizations}
\uses{thm:compactness,def:counting-formulas}
If \(b\notin\operatorname{acl}(B)\), then for every
\(n\in\mathbb N\) there is an elementary extension containing distinct
\(b_0,\ldots,b_n\), each realizing \(\operatorname{tp}(b/B)\).
\end{lemma}

\begin{proof}
Add constants \(b_0,\ldots,b_n\), require each to satisfy the complete
type of \(b\) over \(B\), and require them pairwise distinct.  Every
finite fragment mentions only finitely many formulas from that type;
their conjunction defines an infinite set, since a finite such set would
make \(b\) algebraic over \(B\).  Hence every finite fragment has enough
distinct realizations, and compactness gives the required extension.
\end{proof}

\begin{lemma}[Exchange on a strongly minimal set]
\label{lem:strongly-minimal-exchange}
\uses{def:minimality,def:strongly-minimal-pregeometry,lem:strongly-minimal-closure-laws,lem:nonalgebraic-type-many-realizations,def:counting-formulas}
If \(b\in\operatorname{cl}_A(X\cup\{c\})\setminus\operatorname{cl}_A(X)\),
then \(c\in\operatorname{cl}_A(X\cup\{b\})\).
\end{lemma}

\begin{proof}
Reduce by finite character to finite \(X\), and put
\(B=A\cup X\).  Choose a relation \(R(x,y)\), with parameters from
\(B\), such that \(R(b,c)\) and the fiber \(R(-,c)\) has at most
\(n\) points for some \(n\).  Let
\[
  \chi(y):=(\exists^{\leq n}x)\,R(x,y).
\]
Suppose toward a contradiction that \(c\notin\operatorname{acl}(Bb)\).
Then also \(c\notin\operatorname{acl}(B)\).  Since \(\chi(c)\), the set
\(\chi(D)\) is not finite over \(B\), and strong minimality makes it
cofinite in \(D\).  Likewise the fiber
\(\{y\in D:R(b,y)\}\) contains the nonalgebraic point \(c\), so it is
cofinite.

Because \(b\notin\operatorname{acl}(B)\), choose distinct
\(b_0,\ldots,b_n\) in an elementary extension, all realizing
\(\operatorname{tp}(b/B)\).  The original fiber \(R(b,-)\cap D\) is
infinite, so all counting formulas asserting arbitrarily many points in
that fiber belong to \(\operatorname{tp}(b/B)\).  Hence each
\(R(b_i,-)\cap D\) is infinite and therefore cofinite by strong
minimality.  The finite intersection
\[
  \chi(D)\cap\bigcap_{i\leq n}R(b_i,D)
\]
is nonempty.  Any \(y\) in it satisfies \(\chi(y)\), so at most \(n\)
points satisfy \(R(-,y)\), but the distinct
\(b_0,\ldots,b_n\) all do, a contradiction.  Thus
\(c\in\operatorname{acl}(Bb)\), which is the exchange conclusion.
\end{proof}

\begin{proposition}[Relative algebraic closure is a pregeometry]
\label{prop:strongly-minimal-pregeometry}
\uses{def:strongly-minimal-pregeometry}
For a strongly minimal set \(D\) definable over \(A\),
\(\operatorname{cl}_A\) is a pregeometry on \(D\).
\end{proposition}

\begin{proof}
\uses{lem:strongly-minimal-closure-laws,lem:strongly-minimal-exchange}
Combine the closure laws with exchange.
\end{proof}

\begin{definition}[Independence, bases, and dimension]
\label{def:strongly-minimal-independence}
\uses{def:strongly-minimal-pregeometry,prop:strongly-minimal-pregeometry}
Independence, bases, and dimension are those of the pregeometry
\(\operatorname{cl}_A\).  Write \(\dim_A^M(Y)\) for the cardinality of
a basis of \(Y\subseteq D\) over \(A\).
\end{definition}

\begin{proposition}[Algebraic types are isolated]
\label{prop:algebraic-implies-isolated}
\uses{def:types-over-parameters,def:strongly-minimal-pregeometry}
If \(b\in\operatorname{acl}(A)\), then \(\operatorname{tp}(b/A)\) is
isolated.
\end{proposition}

\begin{proof}
Choose a finite \(A\)-definable set containing \(b\) and refine its
defining formula to distinguish the finitely many possible types.
\end{proof}

\begin{lemma}[Independent tuples in a strongly minimal set have the same type]
\label{lem:independent-tuples-same-type}
\uses{def:minimality,def:strongly-minimal-independence}
Let \(\varphi(v)\) be strongly minimal over \(A\subseteq M_0\), with
\(M_0\preceq M,N\).  Any two finite tuples from \(\varphi(M)\) and
\(\varphi(N)\) that are independent over \(A\) have the same type over
\(A\).
\end{lemma}

\begin{proof}
Induct on tuple length.  At each step a formula true of a new independent
point defines a cofinite subset of the strongly minimal set over the
previous independent tuple.
\end{proof}

\begin{lemma}[A bijection of bases is partial elementary]
\label{lem:basis-bijection-partial-elementary}
\uses{lem:independent-tuples-same-type,def:partial-elementary-embedding}
Let \(\varphi(v)\) be strongly minimal over \(A\subseteq M_0\), with
\(M_0\preceq M,N\).  If \(B\subseteq\varphi(M)\) and
\(C\subseteq\varphi(N)\) are independent over \(A\), then every
bijection \(g:B\to C\) makes \(\operatorname{id}_A\cup g\) a partial
elementary map from \(A\cup B\) to \(A\cup C\).
\end{lemma}

\begin{proof}
Every finite tuple from \(B\) and its image in \(C\) have the same type
over \(A\), which is exactly the finite-parameter criterion for
\(\operatorname{id}_A\cup g\) to be partial elementary.
\end{proof}

\begin{theorem}[Dimension classification for strongly minimal theories]
\label{thm:strongly-minimal-dimension-classification}
\uses{def:minimality,def:strongly-minimal-independence,def:partial-elementary-embedding}
Let \(\varphi(v)\) be a strongly minimal formula over
\(A\subseteq M_0\), where \(M_0\preceq M,N\).  If
\[
  \dim_A^M(\varphi(M))=\dim_A^N(\varphi(N)),
\]
then there is a bijection \(f:\varphi(M)\to\varphi(N)\) such that
\(\operatorname{id}_A\cup f\) is partial elementary.  In particular,
two models of a strongly minimal theory are isomorphic exactly when
they have the same dimension.
\end{theorem}

\begin{proof}
\uses{lem:basis-bijection-partial-elementary,prop:algebraic-implies-isolated,lem:type-space-homeomorphism}
Choose bases and a bijection between them.  By the preceding lemma this
extends the identity on \(A\) to a partial elementary map.  Extend that
map maximally inside the two strongly minimal sets.  Every remaining
point is algebraic over the base together with a finite part of a basis,
so its type over the current domain is isolated.  Transport the isolated
type across the current partial elementary map; its isolating formula is
realized on the other side by elementarity.  This contradicts maximality
unless the map is total, and the symmetric argument makes it surjective.
For a strongly minimal theory take \(A=\emptyset\) and
\(\varphi(v)\equiv(v=v)\).
\end{proof}

\begin{lemma}[Cardinality bound for algebraic closure]
\label{lem:algebraic-closure-cardinality}
\uses{def:strongly-minimal-pregeometry}
In a countable language, for every parameter set \(A\) in a structure,
\[
  |\operatorname{acl}(A)|\leq |A|+\aleph_0.
\]
\end{lemma}

\begin{proof}
There are at most \(|A|+\aleph_0\) formulas with a finite tuple of
parameters from \(A\).  Every algebraic element belongs to one of the
finite sets defined by such a formula, so algebraic closure is a union
of at most \(|A|+\aleph_0\) finite sets.
\end{proof}

\begin{lemma}[Uncountable subsets of a strongly minimal set have full relative dimension]
\label{lem:strongly-minimal-uncountable-subset-full-dimension}
\uses{def:strongly-minimal-independence,lem:algebraic-closure-cardinality}
Let \(D\) be strongly minimal over a countable parameter set \(A\) in a
countable language.  Every uncountable \(Y\subseteq D\) satisfies
\(\dim_A(Y)=|Y|\).
\end{lemma}

\begin{proof}
If \(B\) is a basis of \(Y\), then
\(Y\subseteq\operatorname{acl}(A\cup B)\cap D\).  The algebraic-closure
cardinality bound gives
\(|Y|\leq|A\cup B|+\aleph_0=|B|+\aleph_0\).  Since \(Y\) is uncountable
and \(B\subseteq Y\), this forces \(|B|=|Y|\).
\end{proof}

\begin{theorem}[Omega-stable theories have minimal formulas]
\label{thm:omega-stable-has-minimal-formula}
\uses{def:omega-stability,def:minimality}
Every \(\omega\)-stable theory has a minimal formula in each model.
\end{theorem}

\begin{proof}
\uses{lem:binary-splitting-obstruction}
If no minimal formula existed, every infinite definable set could be
split into two infinite definable subsets, contradicting the binary
splitting obstruction.
\end{proof}

\begin{lemma}[Uniform finite bound without Vaughtian pairs]
\label{lem:finite-bound-without-vaughtian-pairs}
\uses{def:vaughtian-pair}
Suppose \(T\) has no Vaughtian pairs, \(M\models T\), and
\(\varphi(v,\bar w)\) is a formula with fixed parameters from \(M\).
There is \(n\in\mathbb N\) such that, for every \(\bar a\in M\), if
\(|\varphi(M,\bar a)|>n\), then \(\varphi(M,\bar a)\) is infinite.
\end{lemma}

\begin{proof}
\uses{thm:compactness,def:pair-language,lem:pair-relativization,lem:elementary-pair-axiomatization,def:counting-formulas}
Otherwise choose parameters defining arbitrarily large finite sets.
Counting formulas and compactness in the pair language produce a proper
pair-model in which the limiting set is infinite but gains no new points.
The elementary-pair axioms ensure that the \(U\)-part is an elementary
submodel of the ambient reduct, so this pair-model is a Vaughtian pair.
\end{proof}

\begin{theorem}[Minimal formulas are strongly minimal without Vaughtian pairs]
\label{thm:minimal-formula-strongly-minimal}
\uses{def:vaughtian-pair,def:minimality}
If \(T\) has no Vaughtian pairs, every minimal formula is strongly
minimal.
\end{theorem}

\begin{proof}
\uses{lem:finite-bound-without-vaughtian-pairs,def:counting-formulas}
Suppose a minimal formula \(D\) becomes non-minimal in an elementary
extension.  Then for some parameter tuple \(\bar b\), a definable split
of \(D\) has both sides infinite there.  Apply the uniform finite-bound
lemma to the two formulas defining the two sides, obtaining bounds
\(n_0,n_1\).  The extension therefore satisfies the first-order
statement that some parameter tuple makes the first side have more than
\(n_0\) points and the second side more than \(n_1\) points.  By
elementarity the original model has such a parameter tuple.  The two
uniform bounds then force both sides of that split to be infinite in the
original model, contradicting minimality.
\end{proof}

\section{Baldwin--Lachlan and Morley's Theorem}

\begin{lemma}[Prime over an infinite definable set]
\label{lem:prime-over-infinite-definable-set}
\uses{def:omega-stability,def:vaughtian-pair,def:prime-over}
If \(T\) is \(\omega\)-stable with no Vaughtian pairs, \(M\models T\),
\(A\subseteq M\), and \(X\subseteq M\) is infinite and definable over
\(A\), then \(M\) is prime over \(A\cup X\), and no proper elementary
submodel of \(M\) contains \(A\cup X\).
\end{lemma}

\begin{proof}
\uses{thm:prime-extension-over-set}
A proper elementary submodel containing \(A\cup X\) would form a
Vaughtian pair with \(M\).  The prime-extension theorem gives a prime
submodel over \(A\cup X\), which therefore cannot be proper.
\end{proof}

\begin{theorem}[Omega-stability and no Vaughtian pairs imply uncountable categoricity]
\label{thm:omega-stable-no-vaughtian-pairs-implies-uncountably-categorical}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Let \(\kappa\geq\aleph_1\).  If \(T\) is \(\omega\)-stable and has no
Vaughtian pairs, then \(T\) is \(\kappa\)-categorical.
\end{theorem}

\begin{proof}
\uses{thm:prime-extension-over-set,thm:downward-lowenheim-skolem,thm:omega-stable-has-minimal-formula,thm:minimal-formula-strongly-minimal,cor:no-small-infinite-definable-sets-without-vaughtian-pairs,lem:strongly-minimal-uncountable-subset-full-dimension,thm:strongly-minimal-dimension-classification,lem:prime-over-infinite-definable-set}
Let \(M,N\models T\) have size \(\kappa\).  Choose a countable prime
model \(M_0\) and identify elementary copies of it in \(M\) and \(N\).
Choose a minimal formula over parameters from \(M_0\); it is strongly
minimal.  Its realization sets in \(M\) and \(N\) both have size
\(\kappa\), hence both have relative dimension \(\kappa\) over the
whole base \(M_0\).  Apply the relative dimension-classification theorem
with base \(M_0\) to obtain a bijection between those realization sets
whose union with \(\operatorname{id}_{M_0}\) is partial elementary.
Since each ambient model is prime over \(M_0\) together with the infinite
definable set, this partial elementary map extends to an elementary
embedding \(M\to N\); its image contains the whole corresponding
definable set, so it cannot be a proper elementary submodel of \(N\).
Thus the embedding is surjective.
\end{proof}

\begin{theorem}[Baldwin--Lachlan characterization]
\label{thm:baldwin-lachlan-characterization}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
For every uncountable \(\kappa\), \(T\) is \(\kappa\)-categorical if and
only if \(T\) is \(\omega\)-stable and has no Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:uncountable-categoricity-implies-omega-stable-no-vaughtian-pairs,thm:omega-stable-no-vaughtian-pairs-implies-uncountably-categorical}
Combine the two forward milestones with the reverse implication.
\end{proof}

\begin{corollary}[Morley's Categoricity Theorem]
\label{cor:morley-categoricity}
\uses{def:categoricity}
The theory \(T\) is categorical in some uncountable cardinal if and only
if it is categorical in every uncountable cardinal.
\end{corollary}

\begin{proof}
\uses{thm:baldwin-lachlan-characterization}
Categoricity in one uncountable cardinal yields \(\omega\)-stability and
no Vaughtian pairs, and the same characterization then yields
categoricity in every uncountable cardinal.  The converse is immediate.
\end{proof}
