\section{Preliminaries}

Throughout, \(T\) is a complete first-order theory in a countable
language \(L\), and \(T\) has infinite models unless explicitly stated
otherwise.

\begin{definition}[Categoricity in a cardinal]
\label{def:categoricity}
\lean{Cardinal.Categorical}
\mathlibok
The theory \(T\) is \(\kappa\)-categorical if any two models of \(T\)
of cardinality \(\kappa\) are isomorphic.
\end{definition}

\begin{definition}[Embeddings, isomorphisms, and partial embeddings]
\label{def:embeddings}
\lean{FirstOrder.Language.Embedding, FirstOrder.Language.Equiv, FirstOrder.Language.PartialEmbedding}
\leanok
An \(L\)-embedding is an injective map preserving the interpretations of
all function, relation, and constant symbols of \(L\).  An isomorphism
is a bijective \(L\)-embedding.  If \(A\subseteq M\) and \(B\subseteq N\),
a map \(f:A\to B\) is a partial embedding when it preserves all
quantifier-free \(L\)-formulas with parameters from its domain.
\emph{Formalization note.}  The subset version of partial embeddings is
formalized as \texttt{FirstOrder.Language.PartialEmbedding}, with
scoped notation \(A\hookrightarrow_{\!p}[L] B\).  It stores a surjective map
between arbitrary subsets and preservation of quantifier-free formulas;
injectivity is recovered from the equality formula.
\end{definition}

\begin{definition}[Partial elementary embeddings]
\label{def:partial-elementary-embedding}
\lean{FirstOrder.Language.PartialElementaryEmbedding}
\leanok
\uses{def:embeddings}
A partial embedding \(f:A\to B\), where \(A\subseteq M\) and
\(B\subseteq N\), is partial elementary if, for every first-order
formula \(\varphi(\bar v)\) and every tuple \(\bar a\) from \(A\),
\[
  M\models\varphi(\bar a)
  \Longleftrightarrow
  N\models\varphi(f(\bar a)).
\]
\emph{Formalization note.}  The Lean declaration is the one listed
above, with scoped notation \(A\hookrightarrow_{\!pe}[L] B\).  It
stores a surjective map between arbitrary subsets and preservation of
all first-order formulas; its underlying partial embedding is obtained
by \texttt{toPartialEmbedding}.
\end{definition}

\begin{definition}[Elementary embeddings and substructures]
\label{def:elementary-maps}
\lean{FirstOrder.Language.ElementaryEmbedding, FirstOrder.Language.Substructure.IsElementary, FirstOrder.Language.ElementarySubstructure}
\mathlibok
\uses{def:embeddings}
An embedding \(j:M\to N\) is elementary if, for every formula
\(\varphi(\bar v)\) and every tuple \(\bar a\) from \(M\),
\[
  M\models\varphi(\bar a)
  \Longleftrightarrow
  N\models\varphi(j(\bar a)).
\]
If the inclusion \(M\hookrightarrow N\) is elementary, then \(M\preceq N\).
\end{definition}

\begin{theorem}[Tarski--Vaught test]
\label{thm:tarski-vaught-test}
\lean{FirstOrder.Language.Substructure.isElementary_of_exists, FirstOrder.Language.Substructure.toElementarySubstructure}
\mathlibok
\uses{def:elementary-maps}
Let \(M\) be a substructure of \(N\).  Then \(M\preceq N\) if and only
if every existential formula over parameters from \(M\) that has a
witness in \(N\) already has a witness in \(M\).
\emph{Formalization note.}  Mathlib records the witness criterion as a
sufficient condition for elementarity and as a bundled construction of
an elementary substructure.  The converse is immediate from the
definition of elementarity.
\end{theorem}

\begin{definition}[Counting formulas for definable sets]
\label{def:counting-formulas}
\lean{FirstOrder.Language.Formula.iExsAtLeast, FirstOrder.Language.Formula.iExsAtMost, FirstOrder.Language.Formula.iExsExactly, FirstOrder.Language.Formula.existsExactlyLeft, FirstOrder.Language.Formula.realize_iExsAtLeast, FirstOrder.Language.Formula.realize_iExsAtMost, FirstOrder.Language.Formula.realize_iExsExactly, FirstOrder.Language.Formula.realize_existsExactlyLeft}
\leanok
Let \(\varphi(\bar x,\bar y)\) be a formula, where \(\bar x\) is a
finite tuple of object variables, and let \(n\in\mathbb N\).  There are
formulas in the parameter variables \(\bar y\), denoted
\[
  (\exists^{\geq n}\bar x)\,\varphi,\qquad
  (\exists^{\leq n}\bar x)\,\varphi,\qquad
  (\exists^{=n}\bar x)\,\varphi,
\]
asserting respectively that the set defined by
\(\varphi(\bar x,\bar y)\) has at least \(n\), at most \(n\), or
exactly \(n\) elements.
Thus, for every \(L\)-structure \(M\) and parameter tuple \(\bar b\),
\[
\begin{aligned}
M\models(\exists^{\geq n}\bar x)\,\varphi(\bar x,\bar b)
  &\Longleftrightarrow
  \bigl|\{\bar a:M\models\varphi(\bar a,\bar b)\}\bigr|\geq n,\\
M\models(\exists^{\leq n}\bar x)\,\varphi(\bar x,\bar b)
  &\Longleftrightarrow
  \bigl|\{\bar a:M\models\varphi(\bar a,\bar b)\}\bigr|\leq n,\\
M\models(\exists^{=n}\bar x)\,\varphi(\bar x,\bar b)
  &\Longleftrightarrow
  \bigl|\{\bar a:M\models\varphi(\bar a,\bar b)\}\bigr|=n.
\end{aligned}
\]
The first formula existentially quantifies \(n\) pairwise distinct
realizations.  The second is the negation of the existence of
\(n+1\) pairwise distinct realizations, and the third is the
conjunction of the first two.
\end{definition}

\begin{lemma}[Definable-set cardinalities under elementary embeddings]
\label{lem:elementary-embedding-definable-set-cardinality}
\lean{FirstOrder.Language.ElementaryEmbedding.realizations_embedding, FirstOrder.Language.ElementaryEmbedding.mk_realizations_le, FirstOrder.Language.ElementaryEmbedding.encard_realizations_eq_coe_iff, FirstOrder.Language.ElementaryEmbedding.infinite_realizations_iff}
\leanok
\uses{def:elementary-maps}
Let \(e:M\to N\) be an elementary embedding, and let
\[
  X_M=\{\bar a\in M^n:M\models\varphi(\bar a,\bar b)\},\qquad
  X_N=\{\bar c\in N^n:N\models\varphi(\bar c,e\bar b)\}.
\]
Then coordinatewise application of \(e\) embeds \(X_M\) into \(X_N\),
so \(|X_M|\leq |X_N|\).  Moreover, \(X_M\) is finite if and only if
\(X_N\) is finite, and in that case \(|X_M|=|X_N|\).
\end{lemma}

\begin{proof}
\leanok
\uses{def:counting-formulas}
Elementarity preserves every counting formula with the transported
parameters.  Hence the two definable sets have the same exact finite
cardinality whenever either is finite.  The coordinatewise map induced
by \(e\) gives the cardinal inequality without any finiteness
assumption.
\end{proof}

\begin{theorem}[Compactness]
\label{thm:compactness}
\lean{FirstOrder.Language.Theory.isSatisfiable_iff_isFinitelySatisfiable}
\mathlibok
A theory is satisfiable if and only if every finite subset of it is
satisfiable.
\end{theorem}

\begin{theorem}[Downward Loewenheim--Skolem]
\label{thm:downward-lowenheim-skolem}
\lean{FirstOrder.Language.exists_elementarySubstructure_card_eq}
\mathlibok
\uses{def:elementary-maps}
Let \(M\) be an \(L\)-structure and \(X\subseteq M\).  Then \(X\) is
contained in an elementary submodel \(N\preceq M\) whose cardinality is
bounded by \(|X|+|L|+\aleph_0\).
\end{theorem}

\begin{definition}[Elementary chain]
\label{def:elementary-chain}
\lean{FirstOrder.Language.ElementaryChain}
\leanok
\uses{def:elementary-maps}
An (increasing) elementary chain is a family
\((M_i)_{i<\alpha}\) of models of \(T\), indexed by an ordinal
\(\alpha\), whose underlying sets are increasing by inclusion and such
that \(M_i\preceq M_j\) whenever \(i\le j<\alpha\), with
\(M_\lambda=\bigcup_{i<\lambda}M_i\) at each limit ordinal
\(\lambda<\alpha\).
\end{definition}

\begin{theorem}[Tarski's elementary chain theorem]
\label{thm:elementary-chain-union}
\lean{FirstOrder.Language.ElementaryChain.Limit, FirstOrder.Language.ElementaryChain.toLimitElementary, FirstOrder.Language.ElementaryChain.limit_models}
\leanok
\uses{def:elementary-chain}
Let \((M_i)_{i<\alpha}\) be an increasing elementary chain.  Then the
union \(M = \bigcup_{i<\alpha} M_i\) carries a natural
\(L\)-structure, and \(M_i\preceq M\) for every \(i<\alpha\).
\emph{Formalization note.}  The Lean formalization represents the union
as the direct limit of the elementary chain; the canonical maps into this
limit are elementary, and the limit is a model of any theory modeled by
some stage.
\end{theorem}

\begin{proof}
\uses{thm:tarski-vaught-test}
An increasing chain of inclusions ensures that the interpretations of
the symbols of \(L\) cohere, so \(M\) is an \(L\)-structure.  For
elementarity, let \(\exists v\,\varphi(v,\bar a)\) be a formula over
parameters \(\bar a\) from \(M_i\) that holds in \(M\).  A witness
lies in some \(M_j\) with \(j\ge i\); since \(M_i\preceq M_j\),
the Tarski--Vaught test yields a witness already in \(M_i\).
\end{proof}

\begin{lemma}[Cardinality of elementary chain unions]
\label{lem:elementary-chain-cardinality}
\lean{FirstOrder.Language.ElementaryChain.mk_limit_le_of_lift_mk_le, FirstOrder.Language.ElementaryChain.mk_limit_eq_of_forall_lift_mk_eq}
\leanok
\uses{def:elementary-chain}
Let \((M_i)_{i<\alpha}\) be an increasing elementary chain.  If
\(\kappa\) is an infinite cardinal, each \(|M_i|\le\kappa\), and
\(|\alpha|\le\kappa\), then the union has cardinality at most
\(\kappa\).  If instead every \(|M_i|=\kappa\), then the union has
cardinality exactly \(\kappa\).
\end{lemma}

\begin{proof}
The union is a surjective image of the disjoint union
\(\biguplus_{i<\alpha} M_i\).
\end{proof}

\begin{theorem}[Elementary-chain construction principle]
\label{thm:loewenheim-skolem-constructions}
\lean{FirstOrder.Language.ElementaryChain.toLimitElementary, FirstOrder.Language.ElementaryChain.limit_models, FirstOrder.Language.ElementaryChain.mk_limit_le_of_lift_mk_le, FirstOrder.Language.ElementaryChain.mk_limit_eq_of_forall_lift_mk_eq}
\leanok
\uses{def:elementary-chain}
Any inductive construction that builds an increasing elementary chain
of models of \(T\), with cardinality constraints at successor stages,
yields at limit stages a model of \(T\) whose cardinality is controlled
by the chain length and the stage bounds.
\emph{Formalization note.}  The general existence of models in all
sufficiently large cardinalities is supplied by Mathlib's
\texttt{FirstOrder.Language.Theory.exists\_model\_card\_eq}; this node
records only the elementary-chain limit principle used by later
constructions.
\end{theorem}

\begin{proof}
\leanok
\uses{thm:elementary-chain-union,lem:elementary-chain-cardinality}
At a limit stage, take the elementary union of the preceding chain.
Tarski's elementary chain theorem makes this union a model of \(T\),
and the cardinality lemma bounds its size by the cardinality of the
indexing ordinal times the supremum of the stage cardinalities.
\end{proof}

\begin{lemma}[One-step enlargement of parameter-definable sets]
\label{lem:one-step-full-definable-sets}
\lean{FirstOrder.Language.Theory.exists_elementaryExtension_card_eq_with_full_unary_realizations}
\leanok
\uses{def:elementary-maps}
Let \(M\models T\) have infinite cardinality \(\kappa\), with
\(|L|\leq\kappa\).  There are a model \(N\models T\) of cardinality
\(\kappa\) and an elementary embedding \(e:M\to N\) such that every
infinite unary set definable with parameters from \(M\) has
\(\kappa\) realizations in \(N\) after its parameters are transported
along \(e\).
\end{lemma}

\begin{proof}
\leanok
\uses{thm:compactness,thm:downward-lowenheim-skolem}
Add the elementary diagram of \(M\).  For every formula over \(M\)
defining an infinite set, add \(\kappa\) distinct constants satisfying
that formula.  Every finite fragment is realized because the original
set is infinite.  Compactness gives an elementary extension, and
Loewenheim--Skolem gives one of cardinality \(\kappa\).
\end{proof}

\begin{proposition}[Large models with no small infinite definable sets]
\label{prop:large-model-no-small-infinite-definable-sets}
\lean{FirstOrder.Language.Theory.exists_model_of_card_definably_full}
\leanok
Let \(T\) be a theory in a countable language with infinite models, and
let \(\kappa\geq\aleph_1\).  Then \(T\) has a model \(M\) of
cardinality \(\kappa\) such that every parameter-definable subset of
\(M\) is either finite or has cardinality \(\kappa\).
\end{proposition}

\begin{proof}
\leanok
\uses{lem:one-step-full-definable-sets,lem:elementary-embedding-definable-set-cardinality,thm:loewenheim-skolem-constructions}
Start with a model of cardinality \(\kappa\) and iterate the one-step
enlargement along a countable elementary chain.  In the limit, the
finite tuple of parameters defining a unary set comes from some stage.
If the set is infinite in the limit, elementarity and the counting
formulas show that it was already infinite at that stage.  At the next
stage it therefore has \(\kappa\) realizations, which remain distinct in
the limit.  The elementary-chain cardinality bound keeps the limit at
cardinality \(\kappa\).
\end{proof}

\begin{theorem}[Skolem expansion]
\label{thm:skolem-expansion}
For every \(L\)-theory \(T\), there is an expanded language \(L^*\) and
an expanded theory \(T^*\) with built-in Skolem functions.  Each model
of \(T\) admits an expansion to a model of \(T^*\), and
\(|L^*|=|L|+\aleph_0\) may be arranged.
\end{theorem}

\section{Types, Prime Models, and Homogeneity}

\begin{definition}[Types over parameters]
\label{def:types-over-parameters}
\lean{FirstOrder.Language.CompleteTypeOver}
\leanok
\uses{def:elementary-maps}
For \(A\subseteq M\models T\), an \(n\)-type over \(A\) is a set of
\(L_A\)-formulas in \(n\) free variables consistent with
\(\operatorname{Th}_A(M)\).  A complete type decides every such formula,
and \(S_n^M(A)\) denotes the space of complete \(n\)-types over \(A\).
For an \(L_A\)-formula \(\varphi(\bar v)\), write
\[
  [\varphi]=\{p\in S_n^M(A):\varphi\in p\}.
\]
\end{definition}

\begin{definition}[Realization, omission, and isolation]
\label{def:isolated-and-omitted-types}
\lean{FirstOrder.Language.Theory.CompleteType.RealizedBy, FirstOrder.Language.Theory.CompleteType.IsRealizedIn, FirstOrder.Language.Theory.CompleteType.IsOmittedIn, FirstOrder.Language.Theory.CompleteType.IsIsolated, FirstOrder.Language.Theory.CompleteType.IsolatedBy}
\leanok
\uses{def:types-over-parameters}
A tuple realizes a type if it satisfies every formula in that type.  A
model omits a type if no tuple in the model realizes it.  A complete
type \(p\in S_n^M(A)\) is isolated if there is a formula
\(\varphi\in p\) with \([\varphi]=\{p\}\).
\end{definition}

\begin{definition}[Omega-stability]
\label{def:omega-stability}
\lean{FirstOrder.Language.Theory.IsOmegaStable}
\leanok
\uses{def:types-over-parameters}
The theory \(T\) is \(\omega\)-stable if, for every model \(M\models T\),
every countable parameter set \(A\subseteq M\), and every \(n\geq 1\),
the space \(S_n^M(A)\) is countable.
\end{definition}

\begin{lemma}[Splitting a non-isolated basic open set]
\label{lem:isolated-splitting}
\lean{FirstOrder.Language.Theory.CompleteType.exists_isolated_splitting}
\leanok
\uses{def:isolated-and-omitted-types}
Suppose \([\varphi]\subseteq S_n^M(A)\) is nonempty and contains no
isolated type.  Then there is a formula \(\psi\) such that both
\([\varphi\wedge\psi]\) and \([\varphi\wedge\neg\psi]\) are nonempty and
contain no isolated type.
\end{lemma}

\begin{proof}
\leanok
If no such split existed, membership in \([\varphi]\) would determine a
unique complete type, making that type isolated by \(\varphi\).
\end{proof}

\begin{lemma}[Projection of an isolated type]
\label{lem:isolated-projection}
\lean{FirstOrder.Language.Theory.CompleteType.isIsolated_typeOf_left}
\leanok
\uses{def:isolated-and-omitted-types}
If \((\bar a,\bar b)\) realizes an isolated type over \(A\), then
\(\bar a\) realizes an isolated type over \(A\).
\end{lemma}

\begin{proof}
\leanok
An isolating formula \(\theta(\bar v,\bar w)\) for
\(\operatorname{tp}(\bar a,\bar b/A)\) projects to
\(\exists\bar w\,\theta(\bar v,\bar w)\), which isolates the type of
\(\bar a\).
\end{proof}

\begin{lemma}[Transitivity of isolation]
\label{lem:isolated-transitivity}
\lean{FirstOrder.Language.Theory.CompleteType.isIsolated_typeOf_trans}
\leanok
\uses{def:isolated-and-omitted-types}
Let \(M\models T\) and \(A\subseteq B\subseteq M\).  If every finite
tuple from \(B\) realizes an isolated type over \(A\), and \(\bar a\)
realizes an isolated type over \(B\), then \(\bar a\) realizes an
isolated type over \(A\).
\end{lemma}

\begin{proof}
\leanok
\uses{lem:isolated-projection}
Combine an isolating formula for \(\bar a\) over \(B\) with an isolating
formula over \(A\) for the finite tuple of parameters from \(B\) that
occurs in it.  The projected joint type is isolated over \(A\).
\end{proof}

\begin{lemma}[Binary splitting obstruction]
\label{lem:binary-splitting-obstruction}
\lean{FirstOrder.Language.Theory.IsOmegaStable.false_of_formula_splitting}
\leanok
\uses{def:omega-stability}
Let \(T\) be \(\omega\)-stable and \(M\models T\).  Let \(P\) be a property
of formulas over parameters in \(M\).  Suppose \(P\) holds of a root
formula \(\varphi_0\), every formula \(\varphi\) satisfying \(P\) admits
some formula \(\psi\) such that both \(\varphi\wedge\psi\) and
\(\varphi\wedge\neg\psi\) again satisfy \(P\), and every formula
satisfying \(P\) is realizable in \(M\).  Then these assumptions are
inconsistent.
\end{lemma}

\begin{proof}
\leanok
Recursively splitting the root formula gives a binary tree of formulas
satisfying \(P\).  Only countably many parameters occur in this tree, so
it lives over a countable parameter set.  Its nodes define nonempty basic
open sets, each child is contained in its parent, and the two children of
a node are disjoint.  Compactness then gives a point in every branch
intersection, and the \(2^{\aleph_0}\) branches determine pairwise distinct
complete types over the countable parameter set, contradicting
\(\omega\)-stability.
\end{proof}

\begin{theorem}[Density of isolated types in omega-stable theories]
\label{thm:isolated-types-dense}
\lean{FirstOrder.Language.Theory.CompleteType.exists_isolated_mem_typesWith}
\leanok
\uses{def:omega-stability,def:isolated-and-omitted-types}
If \(T\) is \(\omega\)-stable, then for every \(M\models T\) and every
parameter set \(A\subseteq M\), the isolated types are dense in
\(S_n^M(A)\).
\end{theorem}

\begin{proof}
\leanok
\uses{lem:isolated-splitting,lem:binary-splitting-obstruction}
If some nonempty basic open set had no isolated type, repeated splitting
would give a full binary tree of formulas using only countably many
parameters.  Its branches determine \(2^{\aleph_0}\) distinct complete
types over a countable parameter set, contradicting \(\omega\)-stability.
\end{proof}

\begin{lemma}[Type spaces and partial elementary maps]
\label{lem:type-space-homeomorphism}
\lean{FirstOrder.Language.PartialElementaryEmbedding.completeTypeOverHomeomorph}
\leanok
\uses{def:types-over-parameters,def:partial-elementary-embedding}
If \(A\subseteq M\) and \(f:A\to N\) is partial elementary, then
\(S_n^M(A)\) is homeomorphic to \(S_n^N(f(A))\).
\emph{Formalization note.}  The Lean declaration listed above is a
bundled homeomorphism of complete type spaces for an arbitrary type
\(\alpha\) of free variables.  Its codomain is the subset
\(B\subseteq N\) onto which the surjective partial elementary map
\(f:A\hookrightarrow_{\!pe}[L] B\) maps, so \(B\) represents the image
\(f(A)\); the transport acts by renaming the parameter constants along
the equivalence \(A\simeq B\) induced by \(f\).
\end{lemma}

\begin{proof}
\leanok
\uses{thm:compactness}
Apply \(f\) to the parameters occurring in formulas of a type.  Partial
elementarity preserves finite satisfiability, compactness gives a
complete image type, and applying \(f^{-1}\) gives the inverse.  Basic
open sets are carried to basic open sets.
\end{proof}

\begin{definition}[Prime models over parameter sets]
\label{def:prime-over}
\uses{def:elementary-maps,def:partial-elementary-embedding}
A model \(M\models T\) is prime if it elementarily embeds into every
model of \(T\).  If \(A\subseteq M\), then \(M\) is prime over \(A\) if
every partial elementary map from \(A\) into a model of \(T\) extends to
an elementary embedding of \(M\).
\end{definition}

\begin{theorem}[Prime extension over a set]
\label{thm:prime-extension-over-set}
\uses{def:omega-stability,def:prime-over,def:isolated-and-omitted-types}
Let \(T\) be \(\omega\)-stable, let \(M\models T\), and let
\(A\subseteq M\).  There is an elementary submodel \(M_0\preceq M\)
which is prime over \(A\), and \(M_0\) may be chosen so that every
finite tuple from \(M_0\) realizes an isolated type over \(A\).
\end{theorem}

\begin{proof}
\uses{thm:tarski-vaught-test,thm:isolated-types-dense,lem:type-space-homeomorphism,lem:isolated-transitivity}
Build a maximal set \(S\supseteq A\) by repeatedly adjoining elements
whose type over the current set is isolated.  At the stopping stage,
\(S\) is closed under \(L\)-terms: otherwise \(t(\bar s)\notin S\)
would realize the isolated type given by \(x=t(\bar s)\).  Thus \(S\)
is the universe of a substructure.  Density of isolated types and the
Tarski--Vaught test show that this substructure is elementary.  Primeness
is proved by extending a partial elementary map stage by stage; the
homeomorphism of type spaces transports isolating formulas to the image
side, and isolation transitivity keeps the chosen construction isolated
over the original parameter set.
\end{proof}

\begin{definition}[Homogeneous models]
\label{def:homogeneous-model}
\uses{def:partial-elementary-embedding}
Let \(\kappa\) be infinite.  A model \(M\models T\) is
\(\kappa\)-homogeneous if every partial elementary map \(f:A\to M\)
with \(|A|<\kappa\) can be extended over any additional element of
\(M\).  A model is homogeneous if it is \(|M|\)-homogeneous.
\end{definition}

\begin{theorem}[Back-and-forth for countable homogeneous models]
\label{thm:homogeneous-back-and-forth}
\uses{def:homogeneous-model,def:types-over-parameters}
If \(M\) and \(N\) are countable homogeneous models of \(T\) realizing
the same complete \(n\)-types over \(\emptyset\) for every \(n\geq 1\),
then every finite partial elementary map \(f_0:M\rightharpoonup N\)
extends to an isomorphism \(f:M\cong N\).  In particular,
\(M\cong N\).
\end{theorem}

\begin{proof}
Enumerate the two models and perform the standard back-and-forth
construction starting from \(f_0\).  At each one-point extension step,
homogeneity realizes the transported type over the finite current
domain or range.  The union of the finite partial elementary maps is a
bijective elementary map extending \(f_0\).
\end{proof}

\begin{lemma}[Isomorphisms preserve definable set cardinalities]
\label{lem:isomorphism-preserves-definable-set-cardinality}
\uses{def:elementary-maps}
Let \(f:M\cong N\), let \(\varphi(x,\bar y)\) be a formula, and let
\(\bar a\) be a parameter tuple from \(M\).  Then \(f\) restricts to a
bijection
\[
  \varphi(M,\bar a)\cong \varphi(N,f(\bar a)).
\]
\end{lemma}

\begin{proof}
Elementarity gives
\(M\models\varphi(m,\bar a)\) if and only if
\(N\models\varphi(f(m),f(\bar a))\).  Restrict \(f\) and use \(f^{-1}\)
for the inverse.
\end{proof}

\begin{lemma}[Isomorphisms preserve realized type counts]
\label{lem:isomorphism-preserves-realized-type-counts}
\uses{def:types-over-parameters,def:elementary-maps}
If \(f:M\cong N\) and \(A\subseteq M\), then \(f\) induces a bijection
between the complete types over \(A\) realized in \(M\) and the complete
types over \(f(A)\) realized in \(N\).
\end{lemma}

\begin{proof}
Send \(\operatorname{tp}^M(\bar a/A)\) to
\(\operatorname{tp}^N(f(\bar a)/f(A))\).  Elementarity makes this
well-defined, and \(f^{-1}\) gives the inverse.
\end{proof}

\section{Few-Type Models and the Forward Direction}

\begin{definition}[Order indiscernibles]
\label{def:order-indiscernibles}
Let \((I,<)\) be a linear order.  An injective sequence
\((x_i)_{i\in I}\) in a model \(M\) is a sequence of order
indiscernibles if any two increasing finite subsequences of the same
length satisfy the same \(L\)-formulas.
\end{definition}

\begin{theorem}[Existence of order indiscernibles]
\label{thm:existence-of-order-indiscernibles}
\uses{def:order-indiscernibles}
If \(T\) has infinite models and \(I\) is an infinite linear order, then
some model of \(T\) contains an injective sequence of order
indiscernibles indexed by \(I\).
\end{theorem}

\begin{proof}
\uses{thm:compactness}
Use the Ehrenfeucht--Mostowski compactness construction: prescribe the
finite ordered patterns together with the inequalities \(x_i\neq x_j\)
for distinct indices, and apply compactness to obtain an injective
\(I\)-indexed indiscernible sequence.
\end{proof}

\begin{definition}[Skolem hull]
\label{def:skolem-hull}
\uses{thm:skolem-expansion}
If \(M\models T^*\) and \(X\subseteq M\), then \(H(X)\) is the
\(L^*\)-substructure of \(M\) generated by \(X\).  This is the Skolem
hull of \(X\).
\end{definition}

\begin{theorem}[Models realizing few types]
\label{thm:few-types-model}
\uses{def:types-over-parameters}
Let \(L\) be countable and let \(T\) be an \(L\)-theory with infinite
models.  For every \(\kappa\geq\aleph_0\), there is a model
\(M\models T\) of cardinality \(\kappa\) such that, for every
\(A\subseteq M\) and every finite \(n\geq 1\), the model \(M\)
realizes at most \(|A|+\aleph_0\) complete \(n\)-types over \(A\).
\end{theorem}

\begin{proof}
\uses{thm:skolem-expansion,thm:existence-of-order-indiscernibles,def:skolem-hull}
Pass to a countable Skolem expansion.  Index the indiscernibles by the
initial ordinal of cardinality \(\kappa\), with its well-order, and take
their Skolem hull.  Injectivity gives at least \(\kappa\) distinct
generators in the hull, while countability of the Skolem language and
finiteness of terms give at most \(\kappa\) elements; hence the hull has
cardinality exactly \(\kappa\).  A parameter set \(A\) involves only
\(|A|+\aleph_0\) many indices in its Skolem terms; let \(I_A\) be this
set of indices.  For a fixed finite tuple of indices, record its full
quantifier-free order type over \(I_A\): the order and equality relations
among tuple coordinates, and for each coordinate and each \(a\in I_A\),
whether the coordinate is below, equal to, or above \(a\).  Because the
index order is a well-order, the set of elements of \(I_A\) below a
coordinate is an initial segment of \(I_A\), with equality to an element
of \(I_A\) recorded separately.  Hence there are at most
\(|A|+\aleph_0\) such relative order types for any fixed finite tuple.
For a fixed finite tuple of Skolem terms, order indiscernibility then
shows that the relative order type of all indices occurring in those
terms determines the complete type they realize.  Since there are only
countably many Skolem terms, the required bound follows in every finite
arity.
\end{proof}

\begin{lemma}[Non-omega-stability produces a large model realizing many types]
\label{lem:non-omega-stable-large-model-realizing-many-types}
\uses{def:omega-stability,def:types-over-parameters}
If \(T\) is not \(\omega\)-stable and \(\kappa\geq\aleph_1\), then
there is a model \(M\models T\) of cardinality \(\kappa\) and a
countable parameter set \(A\subseteq M\) over which \(M\) realizes
uncountably many complete types.
\end{lemma}

\begin{proof}
\uses{thm:compactness,thm:downward-lowenheim-skolem,thm:loewenheim-skolem-constructions}
Choose uncountably many complete types over a countable parameter set.
Add constants asking for realizations of \(\aleph_1\) of them; finite
satisfiability and compactness give a model realizing them.  Enlarge or
shrink by elementary-chain and Loewenheim--Skolem arguments to obtain
cardinality \(\kappa\).
\end{proof}

\begin{theorem}[Uncountable categoricity implies omega-stability]
\label{thm:uncountable-categoricity-implies-omega-stable}
\uses{def:categoricity,def:omega-stability}
Let \(T\) be a complete theory in a countable language with infinite
models.  If \(T\) is \(\kappa\)-categorical for some
\(\kappa\geq\aleph_1\), then \(T\) is \(\omega\)-stable.
\end{theorem}

\begin{proof}
\uses{lem:non-omega-stable-large-model-realizing-many-types,thm:few-types-model,lem:isomorphism-preserves-realized-type-counts}
If \(T\) were not \(\omega\)-stable, the preceding lemma would give a
model of size \(\kappa\) realizing uncountably many types over a
countable set, while the few-type construction gives a model of size
\(\kappa\) realizing only countably many types over each countable set.
Categoricity and preservation of realized type counts under isomorphism
contradict this.
\end{proof}

\section{Vaughtian Pairs and Two-Cardinal Transfer}

\begin{definition}[\((\kappa,\lambda)\)-models]
\label{def:kappa-lambda-model}
For cardinals \(\kappa>\lambda\geq\aleph_0\), the theory \(T\) has a
\((\kappa,\lambda)\)-model if some model \(M\models T\) has cardinality
\(\kappa\) and some one-variable parameter-definable subset of \(M\) has
cardinality \(\lambda\).
\end{definition}

\begin{definition}[Vaughtian pairs]
\label{def:vaughtian-pair}
\lean{FirstOrder.Language.Theory.IsVaughtianPair, FirstOrder.Language.Theory.HasVaughtianPair}
\leanok
\uses{def:elementary-maps}
A Vaughtian pair for \(T\) is a pair \((N,M)\) of models of \(T\) with
\(M\preceq N\), \(M\neq N\), and a one-variable formula over parameters
from \(M\) whose set of realizations is infinite in \(M\) and unchanged
in \(N\).
\emph{Formalization note.}  Lean represents \(M\preceq N\) by an
elementary embedding \(e:M\hookrightarrow N\), properness by failure of
surjectivity of \(e\), and the unchanged realization set by equality
with the image of the realization set in \(M\).
\end{definition}

\begin{lemma}[\((\kappa,\lambda)\)-models give Vaughtian pairs]
\label{lem:kappa-lambda-model-gives-vaughtian-pair}
\uses{def:kappa-lambda-model,def:vaughtian-pair}
If \(T\) has a \((\kappa,\lambda)\)-model with
\(\kappa>\lambda\geq\aleph_0\), then \(T\) has a Vaughtian pair.
\end{lemma}

\begin{proof}
\uses{thm:downward-lowenheim-skolem}
Let \(N\models T\) have size \(\kappa\), and let \(X=\varphi(N,\bar a)\)
have size \(\lambda\).  Choose \(M\preceq N\) of size \(\lambda\)
containing \(X\) and \(\bar a\).  Then
\(\varphi(M,\bar a)=\varphi(N,\bar a)\), so \((N,M)\) is Vaughtian.
\end{proof}

\begin{corollary}[No small infinite definable sets without Vaughtian pairs]
\label{cor:no-small-infinite-definable-sets-without-vaughtian-pairs}
\uses{def:vaughtian-pair}
Suppose \(T\) has no Vaughtian pairs, \(M\models T\) has cardinality
\(\kappa\), and \(X\subseteq M\) is an infinite parameter-definable set.
Then \(X\) has cardinality \(\kappa\).
\end{corollary}

\begin{proof}
\uses{lem:kappa-lambda-model-gives-vaughtian-pair}
If \(X\) had cardinality \(\lambda\) with
\(\aleph_0\leq\lambda<\kappa\), then \(M\) would be a
\((\kappa,\lambda)\)-model, hence would yield a Vaughtian pair.
\end{proof}

\begin{definition}[The pair language]
\label{def:pair-language}
\uses{def:elementary-maps}
For an elementary pair \(M\preceq N\), let \(L_0=L\cup\{U\}\), where
\(U\) is a unary predicate interpreted as \(M\).  The pair \((N,M)\) is
then treated as a single \(L_0\)-structure.  For each \(L\)-formula
\(\varphi\), the relativization \(\varphi^U\) restricts all quantifiers
to \(U\).
\end{definition}

\begin{lemma}[Countable Vaughtian pair]
\label{lem:countable-vaughtian-pair}
\uses{def:vaughtian-pair,def:pair-language}
If \(T\) is in a countable language and \(T\) has a Vaughtian pair, then
it has a Vaughtian pair \((N_0,M_0)\) with \(N_0\) countable.
\end{lemma}

\begin{proof}
\uses{thm:downward-lowenheim-skolem}
Apply downward Loewenheim--Skolem in the pair language, keeping the
parameters witnessing the Vaughtian pair.  Elementarity of the pair
preserves \(M_0\preceq N_0\), properness, infinitude of the witnessing
definable set, and the assertion that it has no realizations outside
the \(U\)-part.
\end{proof}

\begin{lemma}[Realizing a type on the submodel side]
\label{lem:realize-type-in-submodel-side}
\uses{def:pair-language,def:isolated-and-omitted-types}
Let \(M_0\preceq N_0\) be countable models of \(T\).  If a type over
parameters from \(M_0\) is realized in \(N_0\), then some countable
elementary extension of the pair \((N_0,M_0)\) realizes that type on the
\(M\)-side.
\end{lemma}

\begin{proof}
\uses{thm:compactness,thm:downward-lowenheim-skolem}
Relativize the type to the predicate \(U\) and add it to the elementary
diagram of the pair.  Finite satisfiability follows from
\(M_0\preceq N_0\).  Compactness and Loewenheim--Skolem give a countable
elementary pair extension.  Iterating realizes all countably many
relevant types.
\end{proof}

\begin{lemma}[Realizing a type in an elementary extension of pairs]
\label{lem:realize-type-in-extension-pair}
\uses{def:pair-language,def:isolated-and-omitted-types}
Let \(M_0\preceq N_0\) be countable models of \(T\).  Any type over
parameters from \(N_0\) is realized in some countable elementary
extension of the pair \((N_0,M_0)\).
\end{lemma}

\begin{proof}
\uses{thm:compactness,thm:downward-lowenheim-skolem}
Add the type to the elementary diagram of the pair.  Finite
satisfiability is witnessed in elementary extensions, and compactness
plus Loewenheim--Skolem gives a countable elementary pair extension.
\end{proof}

\begin{lemma}[Homogeneous countable Vaughtian pair]
\label{lem:homogeneous-vaughtian-pair}
\uses{def:vaughtian-pair,def:pair-language,def:homogeneous-model}
Given countable \(M_0\preceq N_0\models T\), there is a countable
elementary extension pair \(M\preceq N\) such that \(M\) and \(N\) are
homogeneous and realize the same complete types over \(\emptyset\).
\end{lemma}

\begin{proof}
\uses{lem:realize-type-in-submodel-side,lem:realize-type-in-extension-pair,thm:loewenheim-skolem-constructions}
Build an elementary chain of pairs.  Repeatedly arrange that types
realized on the larger side are realized on the smaller side, and
arrange the one-point extension requirements for homogeneity on both
sides.  The union is countable and has the stated properties.
\end{proof}

\begin{theorem}[Vaughtian pairs produce an \((\aleph_1,\aleph_0)\)-model]
\label{thm:vaughtian-pair-gives-aleph-one-aleph-zero-model}
\uses{def:vaughtian-pair,def:kappa-lambda-model}
If \(T\) is in a countable language and \(T\) has a Vaughtian pair, then
\(T\) has an \((\aleph_1,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:countable-vaughtian-pair,lem:homogeneous-vaughtian-pair,thm:homogeneous-back-and-forth,thm:loewenheim-skolem-constructions}
Start with a countable homogeneous Vaughtian pair \((N,M)\) whose sides
realize the same complete types, and fix a witnessing formula
\(\varphi(x,\bar a)\) with \(\bar a\in M\).  Build an elementary chain
\((N_\alpha)_{\alpha<\omega_1}\), beginning with \(N_0=N\), such that
each successor pair is isomorphic to \((N,M)\) by an isomorphism whose
restriction to \(M\) fixes \(\bar a\).

At a successor stage, \(M\preceq N_0\preceq N_\alpha\), so the identity
map on the finite tuple \(\bar a\) is partial elementary from \(M\) to
\(N_\alpha\).  The strengthened homogeneous back-and-forth theorem
extends it to an isomorphism \(h_\alpha:M\cong N_\alpha\) fixing
\(\bar a\).  Transporting the extension \(M\preceq N\) along
\(h_\alpha\) gives \(N_{\alpha+1}\succeq N_\alpha\) with
\((N_{\alpha+1},N_\alpha)\cong(N,M)\) over \(\bar a\).  Therefore no
new element of \(N_{\alpha+1}\setminus N_\alpha\) satisfies
\(\varphi(x,\bar a)\).  At a countable limit stage take the elementary
union; it is again countable, homogeneous, and realizes the same
complete types as \(N\), so the construction continues.

The union has size \(\aleph_1\) because every successor extension is
proper, while the fixed definable set
\(\varphi(-,\bar a)\) never grows and hence remains the original
countably infinite set \(\varphi(N_0,\bar a)\).  Thus the union is an
\((\aleph_1,\aleph_0)\)-model.
\end{proof}

\begin{theorem}[Vaught's two-cardinal theorem]
\label{thm:vaught-two-cardinal}
\uses{def:kappa-lambda-model}
If \(T\) is in a countable language and has a \((\kappa,\lambda)\)-model
for some \(\kappa>\lambda\geq\aleph_0\), then \(T\) has an
\((\aleph_1,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:kappa-lambda-model-gives-vaughtian-pair,thm:vaughtian-pair-gives-aleph-one-aleph-zero-model}
The given two-cardinal model yields a Vaughtian pair, and a Vaughtian
pair yields an \((\aleph_1,\aleph_0)\)-model.
\end{proof}

\begin{lemma}[Large definable set with no uncountable definable split]
\label{lem:omega-stable-large-definable-almost-minimal}
\uses{def:omega-stability}
Suppose \(T\) is \(\omega\)-stable, \(M\models T\), and
\(|M|\geq\aleph_1\).  Then there is a formula \(\varphi(v)\) over
parameters from \(M\) such that \(|\varphi(M)|\geq\aleph_1\), and every
parameter-definable split of \(\varphi(M)\) has at least one countable
side.
\end{lemma}

\begin{proof}
\uses{lem:binary-splitting-obstruction}
If every uncountable definable set had an uncountable definable split,
then recursively splitting would produce a binary tree of formulas over
a countable parameter set, hence \(2^{\aleph_0}\) many complete
one-types, contradicting \(\omega\)-stability.
\end{proof}

\begin{lemma}[Proper extension realizing no new countable finite-arity types]
\label{lem:proper-extension-realizes-no-new-countable-types}
\uses{def:omega-stability,def:isolated-and-omitted-types}
Suppose \(T\) is \(\omega\)-stable, \(M\models T\), and
\(|M|\geq\aleph_1\).  Then there is a proper elementary extension
\(N\succeq M\) with \(|N|=|M|\) such that every countable partial type
over \(M\) in a finite tuple of variables that is realized in \(N\) is
already realized in \(M\).
\end{lemma}

\begin{proof}
\uses{lem:omega-stable-large-definable-almost-minimal,thm:prime-extension-over-set,thm:downward-lowenheim-skolem}
Choose the large definable set with no uncountable definable split and
let \(\Gamma\) be the type of its co-countable definable subsets.
Realize \(\Gamma\) by an element \(c\) in an elementary extension of
\(M\).  Downward Loewenheim--Skolem gives an elementary submodel of
cardinality \(|M|\) containing \(M\cup\{c\}\); inside it take a model
prime over \(M\cup\{c\}\).  This prime model is proper over \(M\) and
still has cardinality \(|M|\).  If a countable partial type in a finite
tuple of variables is realized there, isolation of the realizing finite
tuple over \(M\cup\{c\}\) reduces the type to a countable intersection
of co-countable subsets of the large definable set, hence to a
realization already in \(M\).
\end{proof}

\begin{theorem}[Two-cardinal transfer above \(\aleph_1\)]
\label{thm:two-cardinal-transfer}
\uses{def:omega-stability,def:kappa-lambda-model}
Suppose \(T\) is \(\omega\)-stable and has an
\((\aleph_1,\aleph_0)\)-model.  If \(\kappa>\aleph_1\), then \(T\) has
a \((\kappa,\aleph_0)\)-model.
\end{theorem}

\begin{proof}
\uses{lem:proper-extension-realizes-no-new-countable-types,thm:loewenheim-skolem-constructions}
Start from the \((\aleph_1,\aleph_0)\)-model and repeatedly take proper
elementary extensions of the same cardinality as the previous stage,
realizing no new countable finite-arity partial types over that stage.
The one-variable type saying that a new element belongs to the old
countable definable set is countable and therefore remains omitted.  At
limit stages take elementary unions.  The resulting chain of length
\(\kappa\) has union of size \(\kappa\) and keeps that definable set
countable.
\end{proof}

\begin{theorem}[Uncountable categoricity rules out Vaughtian pairs]
\label{thm:uncountable-categoricity-no-vaughtian-pairs}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Let \(T\) be a complete theory in a countable language with infinite
models.  Suppose \(T\) is \(\omega\)-stable and \(T\) is
\(\kappa\)-categorical for some \(\kappa\geq\aleph_1\).  Then \(T\) has
no Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:vaughtian-pair-gives-aleph-one-aleph-zero-model,thm:two-cardinal-transfer,prop:large-model-no-small-infinite-definable-sets,lem:isomorphism-preserves-definable-set-cardinality}
A Vaughtian pair gives an \((\aleph_1,\aleph_0)\)-model, and
two-cardinal transfer gives a \((\kappa,\aleph_0)\)-model when
\(\kappa>\aleph_1\).  Thus there is a model of size \(\kappa\) with an
infinite countable definable set.  The large-model construction gives
another model of size \(\kappa\) in which every infinite definable set
has size \(\kappa\).  Categoricity and preservation of definable set
cardinalities under isomorphism give a contradiction.
\end{proof}

\begin{theorem}[Uncountable categoricity implies omega-stability and no Vaughtian pairs]
\label{thm:uncountable-categoricity-implies-omega-stable-no-vaughtian-pairs}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Let \(T\) be a complete theory in a countable language with infinite
models.  If \(T\) is \(\kappa\)-categorical for some
\(\kappa\geq\aleph_1\), then \(T\) is \(\omega\)-stable and has no
Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:uncountable-categoricity-implies-omega-stable,thm:uncountable-categoricity-no-vaughtian-pairs}
The first theorem gives \(\omega\)-stability.  Applying the second
theorem with this \(\omega\)-stability rules out Vaughtian pairs.
\end{proof}

\section{Strong Minimality and Dimension}

\begin{definition}[Algebraic closure on a strongly minimal set over a base]
\label{def:strongly-minimal-pregeometry}
\uses{def:minimality}
Let \(D\subseteq M\) be a strongly minimal set definable over a
parameter set \(A\subseteq M\).  For \(X\subseteq D\), define
\[
  \operatorname{cl}_A(X)=D\cap\operatorname{acl}(A\cup X).
\]
Thus \(b\in\operatorname{cl}_A(X)\) exactly when \(b\in D\) belongs to
a finite set definable with parameters from \(A\cup X\).
\end{definition}

\begin{proposition}[Relative algebraic closure is a pregeometry]
\label{prop:strongly-minimal-pregeometry}
\uses{def:strongly-minimal-pregeometry}
Let \(D\) be strongly minimal and definable over \(A\).  Then
\(\operatorname{cl}_A\) is a pregeometry on \(D\): it satisfies
reflexivity, monotonicity, idempotence, finite character, and exchange.
\end{proposition}

\begin{proof}
Reflexivity, monotonicity, idempotence, and finite character follow from
the corresponding elementary facts about algebraicity after adjoining
the fixed base \(A\).  Exchange is the standard strongly-minimal
exchange argument over parameters \(A\): if exchange failed, a definable
family of finite fibers would cut \(D\) into an infinite coinfinite
subset definable over parameters, contradicting strong minimality.
\end{proof}

\begin{definition}[Independence, bases, and dimension]
\label{def:strongly-minimal-independence}
\uses{def:strongly-minimal-pregeometry,prop:strongly-minimal-pregeometry}
Let \(D\) be a strongly minimal set in a model \(M\), definable over
\(A\subseteq M\).  A set \(X\subseteq D\) is independent over \(A\) if
no \(x\in X\) lies in
\(\operatorname{cl}_A(X\setminus\{x\})\).  A basis of
\(Y\subseteq D\) over \(A\) is an independent set \(B\subseteq Y\)
such that \(Y\subseteq\operatorname{cl}_A(B)\).  Its cardinality is the
relative dimension \(\dim_A^M(Y)\).  When the ambient model is clear we
write \(\dim_A(Y)\), and when \(A=\emptyset\) simply \(\dim(Y)\).
\end{definition}

\begin{proposition}[Algebraic types are isolated]
\label{prop:algebraic-implies-isolated}
\uses{def:types-over-parameters,def:strongly-minimal-pregeometry}
If \(b\in\operatorname{acl}(A)\), then
\(\operatorname{tp}(b/A)\) is isolated.
\end{proposition}

\begin{proof}
Choose an \(A\)-definable finite set containing \(b\).  Conjoin its
defining formula with formulas distinguishing \(b\) from the finitely
many elements in that set that have different type over \(A\).  The
resulting formula isolates \(\operatorname{tp}(b/A)\).
\end{proof}

\begin{lemma}[Independent tuples in a strongly minimal set have the same type]
\label{lem:independent-tuples-same-type}
\uses{def:minimality,def:strongly-minimal-independence}
Let \(\varphi(v)\) be a strongly minimal formula over parameters
\(A\subseteq M_0\), where \(M_0\models T\) and
\(M_0\preceq M,N\).  If
\(\bar a\in\varphi(M)^n\) and \(\bar b\in\varphi(N)^n\) are
independent over \(A\), then
\[
  \operatorname{tp}^M(\bar a/A)=\operatorname{tp}^N(\bar b/A).
\]
\end{lemma}

\begin{proof}
Induct on \(n\).  In the one-element case, any formula true of one
independent point cuts the strongly minimal set in a cofinite subset,
so it is true of every independent point in any elementary extension
over the base.  The induction step applies the same finite/cofinite
argument over the previous independent tuple.
\end{proof}

\begin{theorem}[Dimension classification for strongly minimal theories]
\label{thm:strongly-minimal-dimension-classification}
\uses{def:minimality,def:strongly-minimal-independence,def:partial-elementary-embedding}
Suppose \(T\) is a strongly minimal theory.  If \(M,N\models T\), then
\(M\cong N\) if and only if \(\dim(M)=\dim(N)\).  More generally, let
\(\varphi(v)\) be a strongly minimal formula over parameters
\(A\subseteq M_0\), where \(M_0\models T\) and \(M_0\preceq M,N\).
If
\[
  \dim_A^M(\varphi(M))=\dim_A^N(\varphi(N)),
\]
then there is a bijective partial elementary map
\(\varphi(M)\to\varphi(N)\) over \(A\).
\end{theorem}

\begin{proof}
\uses{lem:independent-tuples-same-type,prop:algebraic-implies-isolated}
Choose bases \(B\subseteq\varphi(M)\) and
\(C\subseteq\varphi(N)\) over \(A\), and a bijection between them.
Finite tuples from these bases are independent over \(A\), so the
preceding lemma makes this bijection partial elementary over \(A\).
Extend it by maximality.  Since \(B\) and \(C\) are bases over \(A\),
any point outside the current domain or range is algebraic over \(A\)
together with that domain or range.  Its type is therefore isolated and
can be transported across the current partial elementary map and
realized on the other side, contradicting maximality.  Thus the maximal
map is bijective on the strongly minimal sets; in the theory case take
\(A=\emptyset\) to obtain an isomorphism.
\end{proof}

\begin{lemma}[Uncountable subsets of a strongly minimal set have full relative dimension]
\label{lem:strongly-minimal-uncountable-subset-full-dimension}
\uses{def:strongly-minimal-independence}
Let \(D\) be a strongly minimal set definable over a countable parameter
set \(A\) in a model of a countable language, and let
\(Y\subseteq D\) be uncountable.  Then
\[
  \dim_A(Y)=|Y|.
\]
\end{lemma}

\begin{proof}
Let \(B\) be a basis of \(Y\) over \(A\).  By finite character and
countability of the language,
\[
  |\operatorname{acl}(A\cup B)\cap D|\leq |A\cup B|+\aleph_0.
\]
Because \(A\) is countable and
\(Y\subseteq\operatorname{acl}(A\cup B)\cap D\), we get
\(|Y|\leq |B|+\aleph_0\).  Since \(Y\) is uncountable and
\(B\subseteq Y\), this forces \(|B|=|Y|\).
\end{proof}

\begin{theorem}[Omega-stable theories have minimal formulas]
\label{thm:omega-stable-has-minimal-formula}
\uses{def:omega-stability,def:minimality}
Let \(T\) be \(\omega\)-stable.  If \(M\models T\), then there is a
minimal formula in \(M\).
\end{theorem}

\begin{proof}
\uses{lem:binary-splitting-obstruction}
If no minimal formula existed, recursively split each infinite
definable set into two infinite definable subsets.  The resulting
binary tree gives \(2^{\aleph_0}\) complete one-types over a countable
parameter set, contradicting \(\omega\)-stability.
\end{proof}

\begin{lemma}[Uniform finite bound without Vaughtian pairs]
\label{lem:finite-bound-without-vaughtian-pairs}
\uses{def:vaughtian-pair}
Suppose \(T\) has no Vaughtian pairs, \(M\models T\), and
\(\varphi(v,\bar w)\) has parameters from \(M\).  Then there is
\(n\in\mathbb{N}\) such that, whenever \(\bar a\in M\) and
\(|\varphi(M,\bar a)|>n\), the set \(\varphi(M,\bar a)\) is infinite.
\end{lemma}

\begin{proof}
\uses{thm:compactness,def:pair-language,def:counting-formulas}
If no uniform bound existed, choose parameters defining arbitrarily
large finite sets.  Counting formulas express these lower bounds in
first-order logic.  In the pair language, compactness then produces a
proper elementary pair in which the limiting definable set is infinite
but has no new points outside the smaller model.  This is a Vaughtian
pair.
\end{proof}

\begin{theorem}[Minimal formulas are strongly minimal without Vaughtian pairs]
\label{thm:minimal-formula-strongly-minimal}
\uses{def:vaughtian-pair,def:minimality}
If \(T\) has no Vaughtian pairs, then every minimal formula is strongly
minimal.
\end{theorem}

\begin{proof}
\uses{lem:finite-bound-without-vaughtian-pairs}
If a minimal formula became non-minimal in an elementary extension, some
definable subset and its complement would both be infinite there.  The
uniform finite bound transfers this infinitude back to the original
model by elementarity, contradicting minimality.
\end{proof}

\section{Baldwin--Lachlan and Morley's Theorem}

\begin{lemma}[Prime over an infinite definable set]
\label{lem:prime-over-infinite-definable-set}
\uses{def:omega-stability,def:vaughtian-pair,def:prime-over}
If \(T\) is \(\omega\)-stable with no Vaughtian pairs, \(M\models T\),
\(A\subseteq M\), and \(X\subseteq M\) is infinite and definable with
parameters from \(A\), then \(M\) is prime over \(A\cup X\), and no
proper elementary submodel of \(M\) contains \(A\cup X\).
\end{lemma}

\begin{proof}
\uses{thm:prime-extension-over-set}
Write \(X=\varphi(M,\bar a)\) with \(\bar a\) from \(A\).  If a proper
elementary submodel \(N\preceq M\) contained \(A\cup X\), then
\(\bar a\in N\) and elementarity would give
\(\varphi(N,\bar a)=\varphi(M,\bar a)\cap N=X\).  Thus \((M,N)\)
would be a Vaughtian pair, a contradiction.  Hence no such proper
submodel exists.  The prime-extension theorem gives an elementary
submodel of \(M\) prime over \(A\cup X\); since it contains
\(A\cup X\), it cannot be proper, so it is all of \(M\).
\end{proof}

\begin{theorem}[Omega-stability and no Vaughtian pairs imply uncountable categoricity]
\label{thm:omega-stable-no-vaughtian-pairs-implies-uncountably-categorical}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Let \(T\) be a complete theory in a countable language with infinite
models, and let \(\kappa\geq\aleph_1\).  If \(T\) is
\(\omega\)-stable and has no Vaughtian pairs, then \(T\) is
\(\kappa\)-categorical.
\end{theorem}

\begin{proof}
\uses{thm:prime-extension-over-set,thm:downward-lowenheim-skolem,thm:omega-stable-has-minimal-formula,thm:minimal-formula-strongly-minimal,cor:no-small-infinite-definable-sets-without-vaughtian-pairs,lem:strongly-minimal-uncountable-subset-full-dimension,thm:strongly-minimal-dimension-classification,lem:prime-over-infinite-definable-set}
Let \(M,N\models T\) have cardinality \(\kappa\).  By downward
Loewenheim--Skolem choose a countable model of \(T\), and apply the prime
extension theorem over \(\emptyset\) inside it to obtain a countable
prime model \(M_0\).  After identifying its elementary images, regard
\(M_0\preceq M,N\).  Omega-stability gives a minimal formula
\(\varphi(x)\) over \(M_0\), and the absence of Vaughtian pairs makes it
strongly minimal.  The sets \(\varphi(M)\) and \(\varphi(N)\) are
infinite; the no-small-definable-set corollary gives both cardinality
\(\kappa\).  Since \(M_0\) is countable, the full-relative-dimension
lemma gives
\[
  \dim_{M_0}^M(\varphi(M))=\kappa=
  \dim_{M_0}^N(\varphi(N)).
\]
Strongly minimal dimension classification therefore produces a
bijective partial elementary map
\(f:\varphi(M)\to\varphi(N)\) over \(M_0\).

By the preceding lemma, \(M\) is prime over
\(M_0\cup\varphi(M)\), so the partial elementary map consisting of the
identity on \(M_0\) together with \(f\) extends to an elementary
embedding \(F:M\to N\).  Its image \(F(M)\preceq N\) contains
\(M_0\cup\varphi(N)\).  Applying the no-proper-submodel part of the same
lemma inside \(N\) shows \(F(M)=N\).  Thus \(F\) is surjective and
\(M\cong N\).
\end{proof}

\begin{theorem}[Baldwin--Lachlan characterization]
\label{thm:baldwin-lachlan-characterization}
\uses{def:categoricity,def:omega-stability,def:vaughtian-pair}
Let \(T\) be a complete theory in a countable language with infinite
models, and let \(\kappa\geq\aleph_1\).  Then \(T\) is
\(\kappa\)-categorical if and only if \(T\) is \(\omega\)-stable and
has no Vaughtian pairs.
\end{theorem}

\begin{proof}
\uses{thm:uncountable-categoricity-implies-omega-stable-no-vaughtian-pairs,thm:omega-stable-no-vaughtian-pairs-implies-uncountably-categorical}
The forward implication is the uncountable-categoricity theorem above.
The reverse implication is the fixed-cardinal Baldwin--Lachlan theorem.
\end{proof}

\begin{corollary}[Morley's Categoricity Theorem]
\label{cor:morley-categoricity}
\uses{def:categoricity}
Let \(T\) be a complete theory in a countable language with infinite
models.  Then \(T\) is \(\kappa\)-categorical for some uncountable
\(\kappa\) if and only if \(T\) is \(\lambda\)-categorical for every
uncountable \(\lambda\).
\end{corollary}

\begin{proof}
\uses{thm:baldwin-lachlan-characterization}
If \(T\) is categorical in one uncountable cardinal, the
Baldwin--Lachlan characterization gives \(\omega\)-stability and no
Vaughtian pairs, so the same characterization gives categoricity in
every uncountable cardinal.  The converse is immediate by taking
\(\lambda=\aleph_1\).
\end{proof}
